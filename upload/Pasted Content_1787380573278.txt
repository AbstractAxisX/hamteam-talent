/* ========================================================= مدیریت صدا ========================================================= */
var AUDIO_RECS = []; /* {audio, slide, item, keep} */
function pauseAllAudio(){
  AUDIO_RECS.forEach(function(rec){
    if(rec.audio && !rec.audio.paused) rec.audio.pause();
    if(rec.slide) setInlineAudioUI(rec.slide, false);
    if(rec.item) setLbAudioUI(rec.item, false);
  });
}

/* --- پلیر داخل کارت --- */
function ensureInlineAudio(slideEl, m){
  if(slideEl._ar) return slideEl._ar;
  var audio = new Audio(m.src);
  var rec = { audio:audio, slide:slideEl, keep:true };
  slideEl._ar = rec;
  AUDIO_RECS.push(rec);
  audio.addEventListener('loadedmetadata', function(){ slideEl.querySelector('.ad').textContent = fmtT(audio.duration); });
  audio.addEventListener('timeupdate', function(){
    if(audio.duration) slideEl.querySelector('.a-fill').style.width = (audio.currentTime/audio.duration*100) + '%';
    slideEl.querySelector('.ac').textContent = fmtT(audio.currentTime);
  });
  audio.addEventListener('ended', function(){ setInlineAudioUI(slideEl, false); });
  return rec;
}
function setInlineAudioUI(slideEl, playing){
  if(!slideEl || !slideEl.querySelector('.a-play')) return;
  slideEl.classList.toggle('playing', playing);
  var playBtn = slideEl.querySelector('.a-play');
  playBtn.innerHTML = playing ? IC.pause : IC.play;
}
function toggleInlineAudio(slideEl, m){
  var rec = ensureInlineAudio(slideEl, m);
  if(rec.audio.paused){ pauseAllAudio(); rec.audio.play().catch(function(){ showToast('پخش موزیک ناموفق بود 🎧'); }); setInlineAudioUI(slideEl, true); }
  else { rec.audio.pause(); setInlineAudioUI(slideEl, false); }
}
function seekInlineAudio(e, slideEl, m){
  var rec = ensureInlineAudio(slideEl, m);
  if(!rec.audio.duration) return;
  var bar = slideEl.querySelector('.a-bar');
  var r = bar.getBoundingClientRect();
  rec.audio.currentTime = Math.max(0, Math.min(1, (r.right - e.clientX)/r.width)) * rec.audio.duration;
}

    '<span class="s-badge">'+IC.video+'در حال پخش</span>';
}
function stopInlineVideo(slideEl, m){

  slideEl.classList.remove('vid-live');
  slideEl.innerHTML = '<img loading="eager" decoding="async" src="'+m.cover+'" alt="" class="ld">'+
    '<button class="s-play" type="button" aria-label="پخش ویدیو">'+IC.play+'</button>'+
    '<span class="s-badge">'+IC.video+'ویدیو</span><span class="s-dur">'+m.dur+'</span>';
}

document.querySelectorAll('.card').forEach(function(cardEl){
  var p = POSTS.find(function(x){ return x.id === cardEl.getAttribute('data-post'); });
  var car = cardEl.querySelector('.car');
  var track = car.querySelector('.car-track');
  var slides = car.querySelectorAll('.slide');
  var segsBox = car.querySelector('.car-segs');
  var countEl = car.querySelector('.car-count');
  var n = p.media.length;
  var segs = [];
  for(var i = 0; i < n; i++){ var sg = document.createElement('i'); if(i === 0) sg.className = 'on'; segsBox.appendChild(sg); segs.push(sg); }
  var curIdx = 0, ticking = false;

  track.addEventListener('scroll', function(){
    if(ticking) return; ticking = true;
    requestAnimationFrame(function(){
      ticking = false;
      var idx = Math.round(Math.abs(track.scrollLeft) / track.clientWidth);
      idx = Math.max(0, Math.min(n - 1, idx));
      if(idx === curIdx && car.classList.contains('scrolled')) return;
      /* بستن ویدیوی باز مانده در اسلاید قبلی */
      if(idx !== curIdx){
        var prevSlide = slides[curIdx];
        if(prevSlide && prevSlide.classList.contains('vid-live')) stopInlineVideo(prevSlide, p.media[curIdx]);
      }
      curIdx = idx;
      countEl.textContent = faNum(idx + 1) + '/' + faNum(n);
      for(var k = 0; k < n; k++){ segs[k].classList.toggle('on', k <= idx); }
      car.classList.add('scrolled');
      /* پیش‌لود اسلاید بعدی */
      var nxt = slides[idx + 1];
      if(nxt){ var nim = nxt.querySelector('img'); if(nim && nim.loading === 'lazy') nim.loading = 'eager'; }
    });
  }, {passive:true});

  track.addEventListener('click', function(e){
    var slideEl = e.target.closest('.slide');
    if(!slideEl) return;
    var idx = parseInt(slideEl.getAttribute('data-i'), 10);
    var m = p.media[idx];

    if(e.target.closest('.a-play')){ toggleInlineAudio(slideEl, m); return; }
    if(e.target.closest('.a-bar')){ seekInlineAudio(e, slideEl, m); return; }
    if(e.target.closest('.s-play')){ startInlineVideo(slideEl, m); return; }
    if(e.target.closest('.v-x')){ stopInlineVideo(slideEl, m); return; }
    if(e.target.closest('.v-full')){ openLightbox(p, idx); return; }
    if(e.target.closest('.d-view')){ openLightbox(p, idx); return; }

    /* دابل‌تپ = لایک | تک‌تپ = لایت‌باکس */
    if(slideEl._tapTimer){
      clearTimeout(slideEl._tapTimer); slideEl._tapTimer = null;
      burstAt(slideEl, e.clientX, e.clientY);
      likePost(cardEl, true);
    } else {
      slideEl._tapTimer = setTimeout(function(){ slideEl._tapTimer = null; openLightbox(p, idx); }, 260);
    }
  });
});

function burstAt(slideEl, cx, cy){
  var r = slideEl.getBoundingClientRect();
  var b = document.createElement('div');
  b.className = 'burst';
  b.style.left = (cx - r.left) + 'px';
  b.style.top = (cy - r.top) + 'px';
  b.innerHTML = IC.heart.replace('fill="none"', 'fill="#fff" stroke="none"');
  slideEl.appendChild(b);
  setTimeout(function(){ if(b.parentNode) b.remove(); }, 850);
}

function openLightbox(p, startIdx){
  pauseAllAudio();
  lbMediaList = p.media.slice();
  lbTrack.innerHTML = '';
  p.media.forEach(function(m){
    var item = document.createElement('div');
    item.className = 'lb-item';
    item.innerHTML = lbItemHTML(m);
    lbTrack.appendChild(item);
    if(m.t === 'audio') wireLbAudio(item, m);
    if(m.t === 'doc'){
      var pgTrack = item.querySelector('.docv-pages'), pgEl = item.querySelector('.docv-pg'), pn = DOCS[m.key].pages.length;
      pgTrack.addEventListener('scroll', function(){
        var pi = Math.round(Math.abs(pgTrack.scrollLeft)/pgTrack.clientWidth);
        pi = Math.max(0, Math.min(pn - 1, pi));
        pgEl.textContent = 'صفحهٔ ' + faNum(pi + 1) + '/' + faNum(pn);
      }, {passive:true});
      item.querySelector('.docv-dl').addEventListener('click', function(){ showToast('دانلود سند شروع شد ⬇️'); });
    }
  });
  lb.classList.add('open');
  bodyLock(true);
  requestAnimationFrame(function(){
    lbTrack.scrollTo({left: -startIdx * lbTrack.clientWidth, behavior:'auto'});
    syncLb();
  });
  /* پیش‌لود همسایه‌ها */
  [startIdx - 1, startIdx + 1].forEach(function(j){
    var s = p.media[j];
    if(s && s.t === 'image'){ var pre = new Image(); pre.src = s.src; }
  });
}
var lbTick = false;

}, {passive:true});
function syncLb(){
  var n = lbMediaList.length;
  if(!n) return;
  var idx = Math.round(Math.abs(lbTrack.scrollLeft)/lbTrack.clientWidth);
  idx = Math.max(0, Math.min(n - 1, idx));
  lbCount.textContent = faNum(idx + 1) + '/' + faNum(n);
  lbCap.textContent = lbMediaList[idx].title;
}
function closeLightbox(){
  if(!lb.classList.contains('open')) return;
  pauseAllAudio();
  AUDIO_RECS = AUDIO_RECS.filter(function(rec){ return rec.keep; });
  lb.classList.remove('open');
  lbTrack.innerHTML = '';
  bodyLock(false);
}
document.getElementById('lbClose').addEventListener('click', closeLightbox);

var sheet = document.getElementById('sheet'), sheetOverlay = document.getElementById('sheetOverlay');
var shList = document.getElementById('shList'), shTitle = document.getElementById('shTitle');
var shField = document.getElementById('shField'), shSend = document.getElementById('shSend');

var currentPost = null, replyTarget = null;

function openSheet(p){
  currentPost = p;
  clearReply();
  renderComments();
  sheet.style.transform = '';
  sheetOverlay.classList.add('open');
  sheet.classList.add('open');
  bodyLock(true);
}
function closeSheet(){
  if(!sheet.classList.contains('open')) return;
  sheet.classList.remove('open');
  sheetOverlay.classList.remove('open');
  sheet.style.transform = '';
  clearReply();
  shField.blur();
  bodyLock(false);
}

document.getElementById('shClose').addEventListener('click', closeSheet);

/* درگ دستگیره برای بستن */
var shHandle = document.getElementById('shHandle');
var dragStartY = 0, dragDy = 0, dragging = false;
shHandle.addEventListener('pointerdown', function(e){
  dragging = true; dragStartY = e.clientY; dragDy = 0;
  shHandle.setPointerCapture(e.pointerId);
  sheet.style.transition = 'none';
});
shHandle.addEventListener('pointermove', function(e){
  if(!dragging) return;
  dragDy = e.clientY - dragStartY;
  if(dragDy > 0) sheet.style.transform = 'translate(-50%,' + dragDy + 'px)';
});
function dragEnd(){
  if(!dragging) return;
  dragging = false;
  sheet.style.transition = '';
  if(dragDy > 90) closeSheet();
  else sheet.style.transform = '';
  dragDy = 0;
}
shHandle.addEventListener('pointerup', dragEnd);
shHandle.addEventListener('pointercancel', dragEnd);

function clearReply(){
  replyTarget = null;
  shReply.classList.remove('on');
  shField.placeholder = 'نظر خود را بنویسید…';
}

function renderComments(){
  var list = currentPost.comments;
  shTitle.textContent = faNum(currentPost.cCount) + ' نظر';
  shList.innerHTML = '';
  if(!list.length){ shList.innerHTML = '<div class="sh-empty">هنوز نظری ثبت نشده؛ اولین نفر باش ✨</div>'; return; }
  list.forEach(function(cm, ci){
    var node = document.createElement('div');
    node.className = 'cm';
    node.innerHTML =
      '<div class="cm-av" style="background:'+cm.c+'">'+esc(cm.n.charAt(0))+'</div>'+
      '<div class="cm-body">'+
        '<div class="cm-top"><b>'+esc(cm.n)+'</b><span>'+cm.t+'</span></div>'+
        '<div class="cm-txt">'+esc(cm.x)+'</div>'+
        '<div class="cm-act">'+
          '<button class="cm-btn cm-like'+(cm.liked ? ' liked' : '')+'" data-act="clike" data-cid="'+ci+'" data-rid="-1">'+IC.heart+'<span class="cl">'+faNum(cm.likes)+'</span></button>'+
          '<button class="cm-btn" data-act="reply" data-cid="'+ci+'">پاسخ</button>'+
        '</div>'+
        '<div class="cm-replies"></div>'+
      '</div>';
    var rbox = node.querySelector('.cm-replies');
    cm.r.forEach(function(rp, ri){
      var rnode = document.createElement('div');
      rnode.className = 'cm';
      rnode.innerHTML =
        '<div class="cm-av" style="background:'+rp.c+'">'+esc(rp.n.charAt(0))+'</div>'+
        '<div class="cm-body">'+
          '<div class="cm-top"><b>'+esc(rp.n)+'</b><span>'+rp.t+'</span></div>'+
          '<div class="cm-txt">'+esc(rp.x)+'</div>'+
          '<div class="cm-act"><button class="cm-btn cm-like'+(rp.liked ? ' liked' : '')+'" data-act="clike" data-cid="'+ci+'" data-rid="'+ri+'">'+IC.heart+'<span class="cl">'+faNum(rp.likes)+'</span></button></div>'+
        '</div>';
      rbox.appendChild(rnode);
    });
    if(!cm.r.length) rbox.style.display = 'none';
    shList.appendChild(node);
  });
}

shList.addEventListener('click', function(e){
  var btn = e.target.closest('[data-act]');
  if(!btn) return;
  var ci = parseInt(btn.getAttribute('data-cid'), 10);
  var cm = currentPost.comments[ci];
  if(btn.getAttribute('data-act') === 'reply'){
    replyTarget = { ci: ci };
    shReplyName.textContent = cm.n;
    shReply.classList.add('on');
    shField.placeholder = 'پاسخ به ' + cm.n + '…';
    shField.focus();
  } else if(btn.getAttribute('data-act') === 'clike'){
    var ri = parseInt(btn.getAttribute('data-rid'), 10);
    var obj = (ri === -1) ? cm : cm.r[ri];
    obj.liked = !obj.liked;
    obj.likes += obj.liked ? 1 : -1;
    btn.classList.toggle('liked', obj.liked);
    btn.querySelector('.cl').textContent = faNum(obj.likes);
  }
});

function sendComment(){
  var val = shField.value.trim();
  if(!val){ showToast('اول چیزی بنویسید ✍️'); return; }
  var wasReply = !!replyTarget;
  if(wasReply){
    currentPost.comments[replyTarget.ci].r.push({n:'شما', c:'linear-gradient(135deg,#6366f1,#ec4899)', t:'همین حالا', x:val, likes:0, liked:false});
  } else {
    currentPost.comments.push({n:'شما', c:'linear-gradient(135deg,#6366f1,#ec4899)', t:'همین حالا', x:val, likes:0, liked:false, r:[]});
  }
  currentPost.cCount++;
  var cardEl = document.querySelector('.card[data-post="'+currentPost.id+'"]');
  if(cardEl) cardEl.querySelector('.cnt-c').textContent = faNum(currentPost.cCount);
  var keepScroll = shList.scrollTop;
  renderComments();
  shList.scrollTop = wasReply ? shList.scrollHeight : keepScroll;
  clearReply();
  shField.value = '';
  showToast('نظر شما ثبت شد 💬');
}
shSend.addEventListener('click', sendComment);
shField.addEventListener('keydown', function(e){ if(e.key === 'Enter'){ e.preventDefault(); sendComment(); } });

  });
});
