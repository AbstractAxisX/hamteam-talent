    56→function hashIdToIndex(id: string, mod: number): number {
    57→  let h = 0;
    58→  for (let i = 0; i < id.length; i++) {
    59→    h = (h * 31 + id.charCodeAt(i)) >>> 0;
    60→  }
    61→  return h % mod;
    62→}

   382→function ConnectionButton({
   383→  profile,
   384→  onUpdated,
   385→}: {
   386→  profile: ProfileDetail;
   387→  onUpdated: () => void;
   388→}) {
   389→  const [busy, setBusy] = useState(false);
   390→  const status = profile.connectionStatus;
   391→
   392→  async function handle() {
   393→    if (status === "accepted") {
   394→      toast({ title: "شما قبلاً متصل شده‌اید" });
   395→      return;
   396→    }
   397→    setBusy(true);
   398→    try {
   399→      const res = await apiPost<{ status: string }>("/api/connections", {
   400→        receiverId: profile.userId,
   401→      });
   402→      if (res.status === "accepted") {
   403→        toast({ title: "ارتباط برقرار شد ✅" });
   404→      } else if (res.status === "pending-sent") {
   405→        toast({ title: "درخواست ارتباط ارسال شد 📨" });
   406→      }
   407→      onUpdated();
   408→    } catch (e) {
   409→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   410→    } finally {
   411→      setBusy(false);
   412→    }
   413→  }
   414→
   415→  if (status === "accepted") {
   416→    return (
   417→      <Button
   418→        onClick={handle}
   419→        disabled
   420→        className="gap-1.5 rounded-2xl bg-lime/40 text-forest border border-lime/60 font-bold"
   421→      >
   422→        <UserCheck className="w-4 h-4" /> متصل
   423→      </Button>
   424→    );
   425→  }
   426→  if (status === "pending-sent") {
   427→    return (
   428→      <Button
   429→        variant="outline"
   430→        onClick={handle}
   431→        disabled
   432→        className="gap-1.5 rounded-2xl border-gold/40 text-gold font-bold"
   433→      >
   434→        <Clock className="w-4 h-4" /> در انتظار
   435→      </Button>
   436→    );
   437→  }
   438→  if (status === "pending-received") {
   439→    return (
   440→      <Button
   441→        onClick={handle}
   442→        disabled={busy}
   443→        className="gap-1.5 rounded-2xl bg-lime text-forest hover:bg-lime/90 font-bold"
   444→      >
   445→        {busy ? <Loader2 className="w-4 h-4 animate-spin" /> : <UserCheck className="w-4 h-4" />}
   446→        پذیرش درخواست
   447→      </Button>
   448→    );
   449→  }
   450→  return (
   451→    <Button
   452→      onClick={handle}
   453→      disabled={busy}
   454→      className="gap-1.5 rounded-2xl bg-lime text-forest hover:bg-lime/90 font-bold shadow-sm"
   455→    >
   456→      {busy ? <Loader2 className="w-4 h-4 animate-spin" /> : <UserPlus className="w-4 h-4" />}
   457→      دنبال کردن
   458→    </Button>
   459→  );
   460→}

   718→function PostsTab({ userId, isSelf }: { userId: string; isSelf: boolean }) {
   719→  const [posts, setPosts] = useState<PostWithRelations[] | null>(null);
   720→  const [loading, setLoading] = useState(true);
   721→
   722→  const load = useCallback(async () => {
   723→    setLoading(true);
   724→    try {
   725→      const data = await api<{ posts: PostWithRelations[] }>(`/api/posts?userId=${userId}`);
   726→      setPosts(data.posts);
   727→    } catch (e) {
   728→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   729→      setPosts([]);
   730→    } finally {
   731→      setLoading(false);
   732→    }
   733→  }, [userId]);
   734→
   735→  useEffect(() => {
   736→    load();
   737→  }, [load]);
   738→
   739→  if (loading) {
   740→    return (
   741→      <div className="space-y-3">
   742→        {[...Array(2)].map((_, i) => (
   743→          <Card key={i} className="p-4 space-y-3 border-border/60 shadow-card">
   744→            <div className="flex items-center gap-3">
   745→              <Skeleton className="w-11 h-11 rounded-full" />
   746→              <div className="space-y-1.5 flex-1">
   747→                <Skeleton className="h-3.5 w-32 rounded" />
   748→                <Skeleton className="h-2.5 w-20 rounded" />
   749→              </div>
   750→            </div>
   751→            <Skeleton className="h-4 w-full rounded" />
   752→            <Skeleton className="h-4 w-3/4 rounded" />
   753→          </Card>
   754→        ))}
   755→      </div>
   756→    );
   757→  }
   758→
   759→  if (!posts || posts.length === 0) {
   760→    return (
   761→      <Card className="p-5 border-border/60 shadow-card">
   762→        <EmptyState
   763→          kind="posts"
   764→          title="پستی منتشر نشده"
   765→          description={
   766→            isSelf
   767→              ? "پست‌های شما اینجا نمایش داده می‌شوند."
   768→              : "این کاربر هنوز پستی منتشر نکرده است."
   769→          }
   770→          action={
   771→            isSelf ? (
   772→              <Button
   773→                size="sm"
   774→                onClick={() => navigate({ view: "feed" })}
   775→                className="rounded-2xl bg-lime text-forest hover:bg-lime/90 gap-1.5 font-bold"
   776→              >
   777→                <FileText className="w-4 h-4" /> رفتن به فید
   778→              </Button>
   779→            ) : undefined
   780→          }
   781→        />
   782→      </Card>
   783→    );
   784→  }
   785→
   786→  return (
   787→    <div className="space-y-4">
   788→      {posts.map((p, i) => (
   789→        <PostCard key={p.id} post={p} index={i} />
   790→      ))}
   791→    </div>
   792→  );
