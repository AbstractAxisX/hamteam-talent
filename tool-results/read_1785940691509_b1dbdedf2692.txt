     1→"use client";
     2→
     3→import { useEffect, useState, useCallback, useRef, useMemo } from "react";
     4→import { io, type Socket } from "socket.io-client";
     5→import { motion, AnimatePresence } from "framer-motion";
     6→import { api } from "@/lib/api-client";
     7→import { useUser } from "@/lib/use-user";
     8→import { navigate } from "@/lib/nav";
     9→import { Card } from "@/components/ui/card";
    10→import { Button } from "@/components/ui/button";
    11→import { Skeleton } from "@/components/ui/skeleton";
    12→import { Textarea } from "@/components/ui/textarea";
    13→import { EmptyState } from "@/components/shared/empty-state";
    14→import { UserAvatar } from "@/components/shared/user-avatar";
    15→import { toast } from "@/hooks/use-toast";
    16→import { timeAgoFa } from "@/lib/format";
    17→import {
    18→  MessageCircle,
    19→  Send,
    20→  ArrowRight,
    21→  Users,
    22→  Lock,
    23→  Search,
    24→  Sparkles,
    25→} from "lucide-react";
    26→
    27→/* ───────────────────────────── Types ───────────────────────────── */
    28→
    29→type OtherUser = {
    30→  id: string;
    31→  name: string;
    32→  isVerifiedBadge: boolean;
    33→  avatarUrl: string | null;
    34→};
    35→
    36→type ConversationListItem = {
    37→  id: string;
    38→  otherUser: OtherUser;
    39→  lastMessage: {
    40→    content: string;
    41→    createdAt: string;
    42→    senderId: string;
    43→  } | null;
    44→};
    45→
    46→type ChatMessage = {
    47→  id: string;
    48→  conversationId: string;
    49→  senderId: string;
    50→  content: string;
    51→  createdAt: string;
    52→};
    53→
    54→type ConversationDetail = {
    55→  conversation: {
    56→    id: string;
    57→    otherUser: OtherUser & { bioShort?: string };
    58→  } | null;
    59→  messages: ChatMessage[];
    60→};
    61→
    62→/* ───────────────────────────── Main View ───────────────────────────── */
    63→
    64→export function ChatView({ conversationId }: { conversationId?: string }) {
    65→  const { user, loading: userLoading } = useUser();
    66→  const [conversations, setConversations] = useState<ConversationListItem[]>([]);
    67→  const [convLoading, setConvLoading] = useState(true);
    68→  const [activeConv, setActiveConv] = useState<ConversationDetail | null>(null);
    69→  const [msgLoading, setMsgLoading] = useState(false);
    70→  const [draft, setDraft] = useState("");
    71→  const [isTyping, setIsTyping] = useState(false);
    72→  const [search, setSearch] = useState("");
    73→
    74→  const messagesContainerRef = useRef<HTMLDivElement | null>(null);
    75→  const socketRef = useRef<Socket | null>(null);
    76→  const typingTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    77→  const typingClearRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    78→  const activeConvRef = useRef<ConversationDetail | null>(null);
    79→
    80→  /* ── Load conversation list ── */
    81→  const loadConversations = useCallback(async () => {
    82→    setConvLoading(true);
    83→    try {
    84→      const d = await api<{ conversations: ConversationListItem[] }>(
    85→        "/api/chat/conversations"
    86→      );
    87→      setConversations(d.conversations);
    88→    } catch (e) {
    89→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
    90→    } finally {
    91→      setConvLoading(false);
    92→    }
    93→  }, []);
    94→
    95→  /* ── Load messages of active conversation ── */
    96→  const loadMessages = useCallback(async (id: string) => {
    97→    setMsgLoading(true);
    98→    try {
    99→      const d = await api<ConversationDetail>(
   100→        `/api/chat/conversations/${id}/messages`
   101→      );
   102→      setActiveConv(d);
   103→    } catch (e) {
   104→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   105→      setActiveConv(null);
   106→    } finally {
   107→      setMsgLoading(false);
   108→    }
   109→  }, []);
   110→
   111→  useEffect(() => {
   112→    if (user) loadConversations();
   113→    else setConvLoading(false);
   114→  }, [user, loadConversations]);
   115→
   116→  useEffect(() => {
   117→    if (conversationId && user) loadMessages(conversationId);
   118→    else setActiveConv(null);
   119→  }, [conversationId, user, loadMessages]);
   120→
   121→  /* ── Auto-scroll helper ── */
   122→  const scrollToBottom = useCallback(() => {
   123→    const el = messagesContainerRef.current;
   124→    if (el) el.scrollTop = el.scrollHeight;
   125→  }, []);
   126→
   127→  /* ── Keep activeConv ref synced for socket handlers ── */
   128→  useEffect(() => {
   129→    activeConvRef.current = activeConv;
   130→  }, [activeConv]);
   131→
   132→  /* ── Socket.io connection ── */
   133→  useEffect(() => {
   134→    if (!user) return;
   135→    const socket = io("/", {
   136→      path: "/",
   137→      query: { XTransformPort: "3003" },
   138→      auth: { userId: user.id },
   139→      transports: ["websocket", "polling"],
   140→      reconnection: true,
   141→      reconnectionDelay: 1500,
   142→    });
   143→    socketRef.current = socket;
   144→
   145→    socket.on("connect_error", (err: Error) => {
   146→      console.warn("[chat] socket connect_error", err.message);
   147→    });
   148→
   149→    socket.on("message", (msg: ChatMessage) => {
   150→      if (activeConvRef.current?.conversation?.id === msg.conversationId) {
   151→        setActiveConv((prev) => {
   152→          if (!prev) return prev;
   153→          if (prev.messages.some((m) => m.id === msg.id)) return prev;
   154→          return { ...prev, messages: [...prev.messages, msg] };
   155→        });
   156→        queueMicrotask(() => scrollToBottom());
   157→      }
   158→      // Always refresh conversation list to update last message preview
   159→      loadConversations();
   160→    });
   161→
   162→    socket.on(
   163→      "typing",
   164→      (data: { conversationId: string; userId: string; isTyping: boolean }) => {
   165→        if (
   166→          activeConvRef.current?.conversation?.id === data.conversationId &&
   167→          data.userId !== user.id
   168→        ) {
   169→          setIsTyping(data.isTyping);
   170→          if (data.isTyping) {
   171→            if (typingClearRef.current) clearTimeout(typingClearRef.current);
   172→            typingClearRef.current = setTimeout(() => setIsTyping(false), 4000);
   173→          }
   174→        }
   175→      }
   176→    );
   177→
   178→    return () => {
   179→      socket.disconnect();
   180→      socketRef.current = null;
   181→      if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
   182→      if (typingClearRef.current) clearTimeout(typingClearRef.current);
   183→    };
   184→  }, [user, loadConversations, scrollToBottom]);
   185→
   186→  /* ── Emit join when active conversation changes ── */
   187→  useEffect(() => {
   188→    if (!conversationId || !socketRef.current) return;
   189→    if (socketRef.current.connected) {
   190→      socketRef.current.emit("join", { conversationId });
   191→    } else {
   192→      socketRef.current.once("connect", () => {
   193→        socketRef.current?.emit("join", { conversationId });
   194→      });
   195→    }
   196→  }, [conversationId]);
   197→
   198→  /* ── Auto-scroll to bottom on new messages ── */
   199→  useEffect(() => {
   200→    queueMicrotask(() => scrollToBottom());
   201→  }, [activeConv?.messages.length, scrollToBottom]);
   202→
   203→  /* ── Send a message ── */
   204→  const handleSend = useCallback(async () => {
   205→    const content = draft.trim();
   206→    if (!content || !conversationId || !user || !socketRef.current) return;
   207→    setDraft("");
   208→    socketRef.current.emit("typing", {
   209→      conversationId,
   210→      userId: user.id,
   211→      isTyping: false,
   212→    });
   213→    if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
   214→
   215→    const tempId = `temp-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   216→    const optimistic: ChatMessage = {
   217→      id: tempId,
   218→      conversationId,
   219→      senderId: user.id,
   220→      content,
   221→      createdAt: new Date().toISOString(),
   222→    };
   223→    setActiveConv((prev) =>
   224→      prev ? { ...prev, messages: [...prev.messages, optimistic] } : prev
   225→    );
   226→    queueMicrotask(() => scrollToBottom());
   227→
   228→    socketRef.current.emit("message", {
   229→      conversationId,
   230→      senderId: user.id,
   231→      content,
   232→    });
   233→
   234→    setConversations((prev) =>
   235→      prev
   236→        .map((c) =>
   237→          c.id === conversationId
   238→            ? {
   239→                ...c,
   240→                lastMessage: {
   241→                  content,
   242→                  createdAt: optimistic.createdAt,
   243→                  senderId: user.id,
   244→                },
   245→              }
   246→            : c
   247→        )
   248→        .sort((a, b) => {
   249→          const at = a.lastMessage?.createdAt || "0";
   250→          const bt = b.lastMessage?.createdAt || "0";
   251→          return at < bt ? 1 : -1;
   252→        })
   253→    );
   254→  }, [draft, conversationId, user, scrollToBottom]);
   255→
   256→  const handleDraftChange = (val: string) => {
   257→    setDraft(val);
   258→    if (!conversationId || !user || !socketRef.current) return;
   259→    socketRef.current.emit("typing", {
   260→      conversationId,
   261→      userId: user.id,
   262→      isTyping: true,
   263→    });
   264→    if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
   265→    typingTimeoutRef.current = setTimeout(() => {
   266→      socketRef.current?.emit("typing", {
   267→        conversationId,
   268→        userId: user.id,
   269→        isTyping: false,
   270→      });
   271→    }, 1500);
   272→  };
   273→
   274→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   275→    if (e.key === "Enter" && !e.shiftKey) {
   276→      e.preventDefault();
   277→      handleSend();
   278→    }
   279→  };
   280→
   281→  /* ── Filtered list ── */
   282→  const filteredConversations = useMemo(() => {
   283→    if (!search.trim()) return conversations;
   284→    const q = search.trim().toLowerCase();
   285→    return conversations.filter((c) =>
   286→      c.otherUser.name.toLowerCase().includes(q)
   287→    );
   288→  }, [conversations, search]);
   289→
   290→  /* ── Not logged in ── */
   291→  if (!userLoading && !user) {
   292→    return (
   293→      <div className="max-w-3xl mx-auto space-y-5">
   294→        <PageHeader
   295→          icon={MessageCircle}
   296→          title="چت"
   297→          subtitle="گفتگوی زنده با همکاران"
   298→        />
   299→        <Card className="p-0 rounded-2xl border-border/60 shadow-card overflow-hidden">
   300→          <EmptyState
   301→            kind="chat"
   302→            title="برای چت کردن وارد شوید"
   303→            description="برای شروع گفتگو با همکاران، ابتدا وارد حساب کاربری خود شوید."
   304→            action={
   305→              <Button
   306→                onClick={() => navigate({ view: "auth" })}
   307→                className="gap-1.5 rounded-xl"
   308→              >
   309→                <Lock className="w-4 h-4" />
   310→                ورود / ثبت‌نام
   311→              </Button>
   312→            }
   313→          />
   314→        </Card>
   315→      </div>
   316→    );
   317→  }
   318→
   319→  const showMobileList = !conversationId;
   320→
   321→  return (
   322→    <div className="space-y-4">
   323→      <PageHeader
   324→        icon={MessageCircle}
   325→        title="چت"
   326→        subtitle="گفتگوی زنده با همکاران"
   327→      />
   328→
   329→      {/* Mobile: chat list */}
   330→      {showMobileList && (
   331→        <div className="lg:hidden">
   332→          <ChatList
   333→            conversations={filteredConversations}
   334→            loading={convLoading}
   335→            search={search}
   336→            onSearch={setSearch}
   337→            activeId={conversationId}
   338→          />
   339→        </div>
   340→      )}
   341→
   342→      {/* Mobile: active chat */}
   343→      {!showMobileList && (
   344→        <div className="lg:hidden">
   345→          <ChatThread
   346→            conv={activeConv}
   347→            loading={msgLoading}
   348→            draft={draft}
   349→            onDraftChange={handleDraftChange}
   350→            onKeyDown={handleKeyDown}
   351→            onSend={handleSend}
   352→            currentUserId={user?.id || ""}
   353→            isTyping={isTyping}
   354→            messagesContainerRef={messagesContainerRef}
   355→            onBack={() => navigate({ view: "chat" })}
   356→          />
   357→        </div>
   358→      )}
   359→
   360→      {/* Desktop: two-pane — list on right (RTL start), chat on left */}
   361→      <div className="hidden lg:grid lg:grid-cols-[320px_1fr] gap-4">
   362→        <div>
   363→          <ChatList
   364→            conversations={filteredConversations}
   365→            loading={convLoading}
   366→            search={search}
   367→            onSearch={setSearch}
   368→            activeId={conversationId}
   369→          />
   370→        </div>
   371→        <div>
   372→          {conversationId ? (
   373→            <ChatThread
   374→              conv={activeConv}
   375→              loading={msgLoading}
   376→              draft={draft}
   377→              onDraftChange={handleDraftChange}
   378→              onKeyDown={handleKeyDown}
   379→              onSend={handleSend}
   380→              currentUserId={user?.id || ""}
   381→              isTyping={isTyping}
   382→              messagesContainerRef={messagesContainerRef}
   383→            />
   384→          ) : (
   385→            <Card className="h-[70vh] flex items-center justify-center rounded-2xl border-border/60 shadow-card">
   386→              <EmptyState
   387→                kind="chat"
   388→                title="یک گفتگو را انتخاب کنید"
   389→                description="برای شروع گفتگوی جدید، از پروفایل همکاران یا صفحه‌ی افراد استفاده کنید."
   390→                action={
   391→                  <Button
   392→                    variant="outline"
   393→                    onClick={() => navigate({ view: "people" })}
   394→                    className="gap-1.5 rounded-xl"
   395→                  >
   396→                    <Users className="w-4 h-4" />
   397→                    پیدا کردن همکار
   398→                  </Button>
   399→                }
   400→              />
   401→            </Card>
   402→          )}
   403→        </div>
   404→      </div>
   405→    </div>
   406→  );
   407→}
   408→
   409→/* ───────────────────────────── Page Header ───────────────────────────── */
   410→

   412→  icon: Icon,
   413→  title,
   414→  subtitle,

   416→  icon: React.ComponentType<{ className?: string }>;
   417→  title: string;

   424→      </div>
   425→      <div>

   430→      </div>
   431→    </div>
   432→  );

   435→/* ───────────────────────────── Chat List (right pane) ───────────────────────────── */
   436→
   437→function ChatList({
   438→  conversations,
   439→  loading,
   440→  search,
   441→  onSearch,
   442→  activeId,
   443→}: {
   444→  conversations: ConversationListItem[];
   445→  loading: boolean;
   446→  search: string;
   447→  onSearch: (v: string) => void;
   448→  activeId?: string;
   449→}) {
   450→  return (
   451→    <Card className="flex flex-col h-[70vh] overflow-hidden rounded-2xl border-border/60 shadow-card">
   452→      {/* Header */}
   453→      <div className="p-3.5 border-b border-border/60 space-y-2.5 bg-card">
   454→        <div className="flex items-center justify-between gap-2">
   455→          <h2 className="font-bold text-sm">گفتگوها</h2>
   456→          <Button
   457→            variant="ghost"
   458→            size="sm"
   459→            className="h-7 text-xs gap-1 rounded-lg hover:bg-muted"
   460→            onClick={() => navigate({ view: "people" })}
   461→          >
   462→            <Users className="w-3.5 h-3.5" />
   463→            همکاران
   464→          </Button>
   465→        </div>
   466→        <div className="relative">
   467→          <Search className="absolute right-2.5 top-1/2 -translate-y-1/2 w-3.5 h-3.5 text-muted-foreground pointer-events-none" />
   468→          <input
   469→            value={search}
   470→            onChange={(e) => onSearch(e.target.value)}
   471→            placeholder="جست‌وجو در گفتگوها..."
   472→            className="w-full h-9 pr-8 pl-3 text-xs rounded-xl border border-input bg-muted/40 focus:outline-none focus:ring-2 focus:ring-ring/50 focus:border-transparent transition-all"
   473→          />
   474→        </div>
   475→      </div>
   476→
   477→      {/* List */}
   478→      <div className="flex-1 overflow-y-auto slim-scroll">
   479→        {loading ? (
   480→          <div className="p-3 space-y-2">
   481→            {[...Array(5)].map((_, i) => (
   482→              <div key={i} className="flex items-center gap-3 p-2.5 rounded-xl">
   483→                <Skeleton className="w-11 h-11 rounded-full shrink-0" />
   484→                <div className="flex-1 space-y-2">
   485→                  <Skeleton className="h-3.5 w-28 rounded" />
   486→                  <Skeleton className="h-2.5 w-40 rounded" />
   487→                </div>
   488→              </div>
   489→            ))}
   490→          </div>
   491→        ) : conversations.length === 0 ? (
   492→          <div className="p-4">
   493→            <EmptyState
   494→              kind="chat"
   495→              title={search ? "موردی پیدا نشد" : "گفتگویی ندارید"}
   496→              description={
   497→                search
   498→                  ? "عبارت دیگری را امتحان کنید."
   499→                  : "برای شروع گفتگو، به پروفایل یکی از همکاران بروید."
   500→              }
   501→              className="py-6"
   502→              action={
   503→                !search ? (
   504→                  <Button
   505→                    variant="outline"
   506→                    size="sm"
   507→                    onClick={() => navigate({ view: "people" })}
   508→                    className="gap-1.5 rounded-xl"
   509→                  >
   510→                    <Users className="w-4 h-4" />
   511→                    پیدا کردن همکار
   512→                  </Button>
   513→                ) : undefined
   514→              }
   515→            />
   516→          </div>
   517→        ) : (
   518→          conversations.map((c, i) => {
   519→            const active = c.id === activeId;
   520→            return (
   521→              <motion.button
   522→                key={c.id}
   523→                initial={{ opacity: 0, y: 6 }}
   524→                animate={{ opacity: 1, y: 0 }}
   525→                transition={{ duration: 0.25, delay: Math.min(i * 0.03, 0.25) }}
   526→                onClick={() =>
   527→                  navigate({ view: "chat", conversationId: c.id })
   528→                }
   529→                className={`w-full text-right flex items-center gap-3 p-3 hover:bg-muted/40 transition-colors border-b border-border/40 ${
   530→                  active ? "bg-primary/8" : ""
   531→                }`}
   532→              >
   533→                <UserAvatar
   534→                  name={c.otherUser.name}
   535→                  avatarUrl={c.otherUser.avatarUrl}
   536→                  verified={c.otherUser.isVerifiedBadge}
   537→                  size="md"
   538→                />
   539→                <div className="flex-1 min-w-0">
   540→                  <div className="flex items-center justify-between gap-2">
   541→                    <span
   542→                      className={`text-xs truncate ${
   543→                        active
   544→                          ? "font-bold text-primary"
   545→                          : "font-semibold text-foreground"
   546→                      }`}
   547→                    >
   548→                      {c.otherUser.name}
   549→                    </span>
   550→                    {c.lastMessage && (
   551→                      <span className="text-[10px] text-muted-foreground shrink-0">
   552→                        {timeAgoFa(c.lastMessage.createdAt)}
   553→                      </span>
   554→                    )}
   555→                  </div>
   556→                  <p className="text-[11px] text-muted-foreground line-clamp-1 mt-0.5 leading-5">
   557→                    {c.lastMessage
   558→                      ? c.lastMessage.content
   559→                      : "گفتگوی جدید — شروع کنید"}
   560→                  </p>
   561→                </div>
   562→              </motion.button>
   563→            );
   564→          })
   565→        )}
   566→      </div>
   567→    </Card>
   568→  );
   569→}

   571→/* ───────────────────────────── Chat Thread (left pane) ───────────────────────────── */
   572→
   573→function ChatThread({
   574→  conv,
   575→  loading,
   576→  draft,
   577→  onDraftChange,
   578→  onKeyDown,
   579→  onSend,
   580→  currentUserId,
   581→  isTyping,
   582→  messagesContainerRef,
   583→  onBack,
   584→}: {
   585→  conv: ConversationDetail | null;
   586→  loading: boolean;
   587→  draft: string;
   588→  onDraftChange: (v: string) => void;
   589→  onKeyDown: (e: React.KeyboardEvent<HTMLTextAreaElement>) => void;
   590→  onSend: () => void;
   591→  currentUserId: string;
   592→  isTyping: boolean;
   593→  messagesContainerRef: React.RefObject<HTMLDivElement | null>;
   594→  onBack?: () => void;
   595→}) {
   596→  const other = conv?.conversation?.otherUser;
   597→  const messages = conv?.messages ?? [];
   598→
   599→  return (
   600→    <Card className="flex flex-col h-[70vh] overflow-hidden rounded-2xl border-border/60 shadow-card">
   601→      {/* ── Header ── */}
   602→      <div className="p-3 border-b border-border/60 flex items-center gap-3 bg-card">
   603→        {onBack && (
   604→          <Button
   605→            variant="ghost"
