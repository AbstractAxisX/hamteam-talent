    35→    35→/* ───────────────────────────── Types ───────────────────────────── */
    36→    36→
    37→    37→type OtherUser = {
    38→    38→  id: string;
    39→    39→  name: string;
    40→    40→  isVerifiedBadge: boolean;

    45→    45→

    67→    67→
    68→    68→type ConversationDetail = {
    69→    69→  conversation: {
    70→    70→    id: string;

    76→    76→/* ───────────────────────────── Main View ───────────────────────────── */
    77→    77→
    78→    78→export function ChatView({ conversationId }: { conversationId?: string }) {
    79→    79→  const { user, loading: userLoading } = useUser();
    80→    80→  const [conversations, setConversations] = useState<ConversationListItem[]>([]);
    81→    81→  const [requests, setRequests] = useState<ConversationListItem[]>([]);
    82→    82→  const [convLoading, setConvLoading] = useState(true);
    83→    83→  const [activeConv, setActiveConv] = useState<ConversationDetail | null>(null);
    84→    84→  const [msgLoading, setMsgLoading] = useState(false);
    85→    85→  const [draft, setDraft] = useState("");
    86→    86→  const [isTyping, setIsTyping] = useState(false);
    87→    87→  const [search, setSearch] = useState("");
    88→    88→  const [listTab, setListTab] = useState<"messages" | "requests">("messages");
    89→    89→
    90→    90→  const messagesContainerRef = useRef<HTMLDivElement | null>(null);
    91→    91→  const socketRef = useRef<Socket | null>(null);
    92→    92→  const typingTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    93→    93→  const typingClearRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    94→    94→  const activeConvRef = useRef<ConversationDetail | null>(null);
    95→    95→  const activeConvIdRef = useRef<string | undefined>(conversationId);
    96→    96→
    97→    97→  /* ── Load conversation list ── */
    98→    98→  const loadConversations = useCallback(async () => {
    99→    99→    setConvLoading(true);
   100→   100→    try {
   101→   101→      const d = await api<{
   102→   102→        conversations: ConversationListItem[];
   103→   103→        requests: ConversationListItem[];
   104→   104→        unreadCount: number;
   105→   105→      }>("/api/chat/conversations");
   106→   106→      setConversations(d.conversations || []);
   107→   107→      setRequests(d.requests || []);
   108→   108→    } catch (e) {
   109→   109→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   110→   110→    } finally {
   111→   111→      setConvLoading(false);
   112→   112→    }
   113→   113→  }, []);
   114→   114→
   115→   115→  /* ── Load messages of active conversation ── */
   116→   116→  const loadMessages = useCallback(async (id: string) => {
   117→   117→    setMsgLoading(true);
   118→   118→    try {
   119→   119→      const d = await api<ConversationDetail>(
   120→   120→        `/api/chat/conversations/${id}/messages`
   121→   121→      );
   122→   122→      setActiveConv(d);
   123→   123→    } catch (e) {
   124→   124→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   125→   125→      setActiveConv(null);
   126→   126→    } finally {
   127→   127→      setMsgLoading(false);
   128→   128→    }
   129→   129→  }, []);
   130→   130→
   131→   131→  useEffect(() => {
   132→   132→    if (user) loadConversations();
   133→   133→    else setConvLoading(false);
   134→   134→  }, [user, loadConversations]);
   135→   135→
   136→   136→  useEffect(() => {
   137→   137→    activeConvIdRef.current = conversationId;
   138→   138→    if (conversationId && user) {
   139→   139→      loadMessages(conversationId);
   140→   140→      // Mark messages as read
   141→   141→      apiPost(`/api/chat/conversations/${conversationId}/read`)
   142→   142→        .then(() => loadConversations())
   143→   143→        .catch(() => {});
   144→   144→    } else {
   145→   145→      setActiveConv(null);
   146→   146→    }
   147→   147→  }, [conversationId, user, loadMessages, loadConversations]);
   148→   148→
   149→   149→  /* ── Auto-scroll helper ── */
   150→   150→  const scrollToBottom = useCallback(() => {
   151→   151→    const el = messagesContainerRef.current;
   152→   152→    if (el) el.scrollTop = el.scrollHeight;
   153→   153→  }, []);
   154→   154→
   155→   155→  /* ── Keep activeConv ref synced for socket handlers ── */
   156→   156→  useEffect(() => {
   157→   157→    activeConvRef.current = activeConv;
   158→   158→  }, [activeConv]);
   159→   159→
   160→   160→  /* ── Socket.io connection (mini-service on port 3003 via gateway) ── */
   161→   161→  useEffect(() => {
   162→   162→    if (!user) return;
   163→   163→    const socket = io("/", {
   164→   164→      path: "/",
   165→   165→      query: { XTransformPort: "3003" },
   166→   166→      auth: { userId: user.id },
   167→   167→      transports: ["websocket", "polling"],
   168→   168→      reconnection: true,
   169→   169→      reconnectionDelay: 1500,
   170→   170→    });
   171→   171→    socketRef.current = socket;
   172→   172→
   173→   173→    socket.on("connect_error", (err: Error) => {
   174→   174→      console.warn("[chat] socket connect_error", err.message);
   175→   175→    });
   176→   176→
   177→   177→    socket.on("message", (msg: ChatMessage) => {
   178→   178→      if (activeConvRef.current?.conversation?.id === msg.conversationId) {
   179→   179→        setActiveConv((prev) => {
   180→   180→          if (!prev) return prev;
   181→   181→          if (prev.messages.some((m) => m.id === msg.id)) return prev;
   182→   182→          return { ...prev, messages: [...prev.messages, msg] };
   183→   183→        });
   184→   184→        queueMicrotask(() => scrollToBottom());
   185→   185→        // Mark read if I'm the receiver
   186→   186→        apiPost(`/api/chat/conversations/${msg.conversationId}/read`).catch(() => {});
   187→   187→      }
   188→   188→      // Always refresh conversation list to update last message preview + unread counts
   189→   189→      loadConversations();
   190→   190→    });
   191→   191→
   192→   192→    socket.on(
   193→   193→      "typing",
   194→   194→      (data: { conversationId: string; userId: string; isTyping: boolean }) => {
   195→   195→        if (
   196→   196→          activeConvRef.current?.conversation?.id === data.conversationId &&
   197→   197→          data.userId !== user.id
   198→   198→        ) {
   199→   199→          setIsTyping(data.isTyping);
   200→   200→          if (data.isTyping) {
   201→   201→            if (typingClearRef.current) clearTimeout(typingClearRef.current);
   202→   202→            typingClearRef.current = setTimeout(() => setIsTyping(false), 4000);
   203→   203→          }
   204→   204→        }
   205→   205→      }
   206→   206→    );
   207→   207→
   208→   208→    return () => {
   209→   209→      socket.disconnect();
   210→   210→      socketRef.current = null;
   211→   211→      if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
   212→   212→      if (typingClearRef.current) clearTimeout(typingClearRef.current);
   213→   213→    };
   214→   214→  }, [user, loadConversations, scrollToBottom]);
   215→   215→
   216→   216→  /* ── Emit join when active conversation changes ── */
   217→   217→  useEffect(() => {
   218→   218→    if (!conversationId || !socketRef.current) return;
   219→   219→    if (socketRef.current.connected) {
   220→   220→      socketRef.current.emit("join", { conversationId });
   221→   221→    } else {
   222→   222→      socketRef.current.once("connect", () => {
   223→   223→        socketRef.current?.emit("join", { conversationId });
   224→   224→      });
   225→   225→    }
   226→   226→  }, [conversationId]);
   227→   227→
   228→   228→  /* ── Auto-scroll to bottom on new messages ── */
   229→   229→  useEffect(() => {
   230→   230→    queueMicrotask(() => scrollToBottom());
   231→   231→  }, [activeConv?.messages.length, scrollToBottom]);
   232→   232→
   233→   233→  /* ── Send a message ── */
   234→   234→  const handleSend = useCallback(() => {
   235→   235→    const content = draft.trim();
   236→   236→    if (!content || !conversationId || !user || !socketRef.current) return;
   237→   237→    setDraft("");
   238→   238→    socketRef.current.emit("typing", {
   239→   239→      conversationId,
   240→   240→      userId: user.id,
   241→   241→      isTyping: false,
   242→   242→    });
   243→   243→    if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
   244→   244→
   245→   245→    const tempId = `temp-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   246→   246→    const optimistic: ChatMessage = {
   247→   247→      id: tempId,
   248→   248→      conversationId,
   249→   249→      senderId: user.id,
   250→   250→      content,
   251→   251→      createdAt: new Date().toISOString(),
   252→   252→    };
   253→   253→    setActiveConv((prev) =>
   254→   254→      prev ? { ...prev, messages: [...prev.messages, optimistic] } : prev
   255→   255→    );
   256→   256→    queueMicrotask(() => scrollToBottom());
   257→   257→
   258→   258→    socketRef.current.emit("message", {
   259→   259→      conversationId,
   260→   260→      senderId: user.id,
   261→   261→      content,
   262→   262→    });
   263→   263→
   264→   264→    setConversations((prev) =>
   265→   265→      prev
   266→   266→        .map((c) =>
   267→   267→          c.id === conversationId
   268→   268→            ? {
   269→   269→                ...c,
   270→   270→                lastMessage: {
   271→   271→                  content,
   272→   272→                  createdAt: optimistic.createdAt,
   273→   273→                  senderId: user.id,
   274→   274→                },
   275→   275→              }
   276→   276→            : c
   277→   277→        )
   278→   278→        .sort((a, b) => {
   279→   279→          const at = a.lastMessage?.createdAt || "0";
   280→   280→          const bt = b.lastMessage?.createdAt || "0";
   281→   281→          return at < bt ? 1 : -1;
   282→   282→        })
   283→   283→    );
   284→   284→  }, [draft, conversationId, user, scrollToBottom]);
   285→   285→
   286→   286→  const handleDraftChange = (val: string) => {
   287→   287→    setDraft(val);
   288→   288→    if (!conversationId || !user || !socketRef.current) return;
   289→   289→    socketRef.current.emit("typing", {
   290→   290→      conversationId,
   291→   291→      userId: user.id,
   292→   292→      isTyping: true,
   293→   293→    });
   294→   294→    if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
   295→   295→    typingTimeoutRef.current = setTimeout(() => {
   296→   296→      socketRef.current?.emit("typing", {
   297→   297→        conversationId,
   298→   298→        userId: user.id,
   299→   299→        isTyping: false,
   300→   300→      });
   301→   301→    }, 1500);
   302→   302→  };
   303→   303→
   304→   304→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   305→   305→    if (e.key === "Enter" && !e.shiftKey) {
   306→   306→      e.preventDefault();
   307→   307→      handleSend();
   308→   308→    }
   309→   309→  };
   310→   310→
   311→   311→  /* ── Active conversation status (from list) ── */
   312→   312→  const activeConvInfo = useMemo(() => {
   313→   313→    if (!conversationId) return null;
   314→   314→    const inList =
   315→   315→      conversations.find((c) => c.id === conversationId) ||
   316→   316→      requests.find((c) => c.id === conversationId);
   317→   317→    if (inList) return inList;
   318→   318→    // Default: assume active when not found in either list
   319→   319→    return {
   320→   320→      id: conversationId,
   321→   321→      otherUser: activeConv?.conversation?.otherUser ?? {
   322→   322→        id: "",
   323→   323→        name: "",
   324→   324→        isVerifiedBadge: false,
   325→   325→        avatarUrl: null,
   326→   326→      },
   327→   327→      lastMessage: null,
   328→   328→      status: "active" as const,
   329→   329→      initiatorId: null,
   330→   330→      unreadCount: 0,
   331→   331→    };
   332→   332→  }, [conversationId, conversations, requests, activeConv]);
   333→   333→
   334→   334→  /* ── Filtered list ── */
   335→   335→  const filteredConversations = useMemo(() => {
   336→   336→    if (!search.trim()) return conversations;
   337→   337→    const q = search.trim().toLowerCase();
   338→   338→    return conversations.filter((c) =>
   339→   339→      c.otherUser.name.toLowerCase().includes(q)
   340→   340→    );
   341→   341→  }, [conversations, search]);
   342→   342→
   343→   343→  const filteredRequests = useMemo(() => {
   344→   344→    if (!search.trim()) return requests;
   345→   345→    const q = search.trim().toLowerCase();
   346→   346→    return requests.filter((c) =>
   347→   347→      c.otherUser.name.toLowerCase().includes(q)
   348→   348→    );
   349→   349→  }, [requests, search]);
   350→   350→
   351→   351→  /* ── Not logged in ── */
   352→   352→  if (!userLoading && !user) {
   353→   353→    return (
   354→   354→      <div className="max-w-3xl mx-auto space-y-5">
   355→   355→        <PageHeader unreadCount={0} requestsCount={0} />
   356→   356→        <Card className="p-0 rounded-2xl border-border/60 overflow-hidden">
   357→   357→          <EmptyState
   358→   358→            kind="chat"
   359→   359→            title="برای چت کردن وارد شوید"
   360→   360→            description="برای شروع گفتگو با همکاران، ابتدا وارد حساب کاربری خود شوید."
   361→   361→            action={
   362→   362→              <Button
   363→   363→                onClick={() => navigate({ view: "auth" })}
   364→   364→                className="gap-1.5 rounded-2xl bg-primary text-primary-foreground font-bold hover:bg-primary/90"
   365→   365→              >
   366→   366→                <Lock className="w-4 h-4" />
   367→   367→                ورود / ثبت‌نام
   368→   368→              </Button>
   369→   369→            }
   370→   370→          />
   371→   371→        </Card>
   372→   372→      </div>
   373→   373→    );
   374→   374→  }
   375→   375→
   376→   376→  const showMobileList = !conversationId;
   377→   377→
   378→   378→  return (
   379→   379→    <div className="space-y-4">
   380→   380→      <PageHeader
   381→   381→        unreadCount={conversations.reduce((s, c) => s + (c.unreadCount || 0), 0)}
   382→   382→        requestsCount={requests.length}
   383→   383→      />
   384→   384→
   385→   385→      {/* Mobile: chat list OR active chat */}
   386→   386→      {showMobileList && (
   387→   387→        <div className="lg:hidden">
   388→   388→          <ChatListPanel
   389→   389→            conversations={filteredConversations}
   390→   390→            requests={filteredRequests}
   391→   391→            loading={convLoading}
   392→   392→            search={search}
   393→   393→            onSearch={setSearch}
   394→   394→            activeId={conversationId}
   395→   395→            listTab={listTab}
   396→   396→            onTabChange={setListTab}
   397→   397→            onRespond={async (id, action) => {
   398→   398→              try {
   399→   399→                await apiPost(`/api/chat/conversations/${id}/respond`, { action });
   400→   400→                toast({
   401→   401→                  title: action === "accept" ? "درخواست تأیید شد ✅" : "درخواست رد شد",
   402→   402→                });
   403→   403→                await loadConversations();
   404→   404→              } catch (e) {
   405→   405→                toast({
   406→   406→                  title: "خطا",
   407→   407→                  description: (e as Error).message,
   408→   408→                  variant: "destructive",
   409→   409→                });
   410→   410→              }
   411→   411→            }}
   412→   412→          />
   413→   413→        </div>
   414→   414→      )}
   415→   415→
   416→   416→      {!showMobileList && (
   417→   417→        <div className="lg:hidden">
   418→   418→          <ChatThread
   419→   419→            conv={activeConv}
   420→   420→            convInfo={activeConvInfo}
   421→   421→            loading={msgLoading}
   422→   422→            draft={draft}
   423→   423→            onDraftChange={handleDraftChange}
   424→   424→            onKeyDown={handleKeyDown}
   425→   425→            onSend={handleSend}
   426→   426→            currentUserId={user?.id || ""}
   427→   427→            isTyping={isTyping}
   428→   428→            messagesContainerRef={messagesContainerRef}
   429→   429→            onBack={() => navigate({ view: "chat" })}
   430→   430→            onRespond={async (id, action) => {
   431→   431→              try {
   432→   432→                await apiPost(`/api/chat/conversations/${id}/respond`, { action });
   433→   433→                toast({
   434→   434→                  title: action === "accept" ? "درخواست تأیید شد ✅" : "درخواست رد شد",
   435→   435→                });
   436→   436→                if (action === "accept") {
   437→   437→                  await loadMessages(id);
   438→   438→                  await loadConversations();
   439→   439→                } else {
   440→   440→                  navigate({ view: "chat" });
   441→   441→                }
   442→   442→              } catch (e) {
   443→   443→                toast({
   444→   444→                  title: "خطا",
   445→   445→                  description: (e as Error).message,
   446→   446→                  variant: "destructive",
   447→   447→                });
   448→   448→              }
   449→   449→            }}
   450→   450→          />
   451→   451→        </div>
   452→   452→      )}
   453→   453→
   454→   454→      {/* Desktop: two-pane — list on right (RTL start), chat on left */}
   455→   455→      <div className="hidden lg:grid lg:grid-cols-[1fr_340px] gap-4">
   456→   456→        <div>
   457→   457→          {conversationId ? (
   458→   458→            <ChatThread
   459→   459→              conv={activeConv}
   460→   460→              convInfo={activeConvInfo}
   461→   461→              loading={msgLoading}
   462→   462→              draft={draft}
   463→   463→              onDraftChange={handleDraftChange}
   464→   464→              onKeyDown={handleKeyDown}
   465→   465→              onSend={handleSend}
   466→   466→              currentUserId={user?.id || ""}
   467→   467→              isTyping={isTyping}
   468→   468→              messagesContainerRef={messagesContainerRef}
   469→   469→              onRespond={async (id, action) => {
   470→   470→                try {
   471→   471→                  await apiPost(`/api/chat/conversations/${id}/respond`, { action });
   472→   472→                  toast({
   473→   473→                    title: action === "accept" ? "درخواست تأیید شد ✅" : "درخواست رد شد",
   474→   474→                  });
   475→   475→                  if (action === "accept") {
   476→   476→                    await loadMessages(id);
   477→   477→                    await loadConversations();
   478→   478→                  } else {
   479→   479→                    navigate({ view: "chat" });
   480→   480→                  }
   481→   481→                } catch (e) {
   482→   482→                  toast({
   483→   483→                    title: "خطا",
   484→   484→                    description: (e as Error).message,
   485→   485→                    variant: "destructive",
   486→   486→                  });
   487→   487→                }
   488→   488→              }}
   489→   489→            />
   490→   490→          ) : (
   491→   491→            <Card className="h-[72vh] flex items-center justify-center rounded-2xl border-border/60 bg-muted/30">
   492→   492→              <EmptyState
