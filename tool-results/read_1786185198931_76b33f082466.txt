    67→function hashIdToIndex(id: string, mod: number): number {
    68→  let h = 0;
    69→  for (let i = 0; i < id.length; i++) {
    70→    h = (h * 31 + id.charCodeAt(i)) >>> 0;
    71→  }
    72→  return h % mod;
    73→}

   242→function PhotosBioSection({
   243→  profile,
   244→  onUpdated,
   245→}: {
   246→  profile: ProfileDetail;
   247→  onUpdated: () => void;
   248→}) {
   249→  const [avatarUrl, setAvatarUrl] = useState(profile.avatarUrl ?? "");
   250→  const [bannerUrl, setBannerUrl] = useState(profile.bannerUrl ?? "");
   251→  const [bioShort, setBioShort] = useState(profile.bioShort);
   252→  const [bioLong, setBioLong] = useState(profile.bioLong);
   253→  const [saving, setSaving] = useState(false);
   254→
   255→  useEffect(() => {
   256→    setAvatarUrl(profile.avatarUrl ?? "");
   257→    setBannerUrl(profile.bannerUrl ?? "");
   258→    setBioShort(profile.bioShort);
   259→    setBioLong(profile.bioLong);
   260→  }, [profile]);
   261→
   262→  const isDefaultBanner = !bannerUrl || bannerUrl.startsWith("default");
   263→  const defaultIdx = isDefaultBanner
   264→    ? bannerUrl
   265→      ? Math.max(0, Number(bannerUrl.replace("default-", "")) - 1)
   266→      : hashIdToIndex(profile.userId, BANNER_GRADIENTS.length)
   267→    : 0;
   268→
   269→  async function save() {
   270→    setSaving(true);
   271→    try {
   272→      await apiPut("/api/profile/me", {
   273→        avatarUrl: avatarUrl.trim() || null,
   274→        bannerUrl: bannerUrl.trim() || null,
   275→        bioShort,
   276→        bioLong,
   277→      });
   278→      toast({ title: "ذخیره شد ✅" });
   279→      onUpdated();
   280→    } catch (e) {
   281→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   282→    } finally {
   283→      setSaving(false);
   284→    }
   285→  }
   286→
   287→  return (
   288→    <Card className="p-5 md:p-6 space-y-5 border-border/60 shadow-card rounded-2xl">
   289→      <SectionTitle icon={ImageIcon} title="عکس‌ها و بیو" />
   290→
   291→      {/* Banner preview */}
   292→      <div className="space-y-2">
   293→        <Label>پیش‌نمایش بنر</Label>
   294→        <div className="h-28 md:h-32 rounded-2xl overflow-hidden relative border border-border/60">
   295→          {isDefaultBanner ? (
   296→            <div
   297→              className="absolute inset-0"
   298→              style={{ background: BANNER_GRADIENTS[defaultIdx] }}

   302→            </div>
   303→          ) : (
   304→            <img src={bannerUrl} alt="بنر" className="absolute inset-0 w-full h-full object-cover" />
   305→          )}
   306→        </div>
   307→        <RadioGroup
   308→          value={isDefaultBanner ? `default-${defaultIdx + 1}` : "custom"}
   309→          onValueChange={(v) => {
   310→            if (v === "custom") {
   311→              setBannerUrl("");
   312→            } else {
   313→              setBannerUrl(v);
   314→            }
   315→          }}
   316→          className="grid grid-cols-3 sm:grid-cols-7 gap-2"
   317→        >
   318→          {BANNER_GRADIENTS.map((g, i) => (
   319→            <Label
   320→              key={i}
   321→              htmlFor={`banner-${i}`}
   322→              className="cursor-pointer flex flex-col items-center gap-1"
   323→            >
   324→              <RadioGroupItem id={`banner-${i}`} value={`default-${i + 1}`} className="sr-only" />
   325→              <div
   326→                className={cn(
   327→                  "h-10 w-full rounded-lg border-2",
   328→                  isDefaultBanner && defaultIdx === i
   329→                    ? "border-forest shadow-md scale-105"
   330→                    : "border-transparent"
   331→                )}
   332→                style={{ background: g }}
   333→              />
   334→              <span className="text-[10px] text-muted-foreground nums-fa">{toFa(i + 1)}</span>
   335→            </Label>
   336→          ))}
   337→          <Label htmlFor="banner-custom" className="cursor-pointer flex flex-col items-center gap-1">
   338→            <RadioGroupItem id="banner-custom" value="custom" className="sr-only" />

   340→              className={cn(
   341→                "h-10 w-full rounded-lg border-2 grid place-items-center bg-muted",

   351→          <Input
   352→            placeholder="آدرس تصویر بنر (URL)"
   353→            value={bannerUrl}
   354→            onChange={(e) => setBannerUrl(e.target.value)}
   355→            dir="ltr"
   356→            className="rounded-2xl"
   357→          />
   358→        )}
