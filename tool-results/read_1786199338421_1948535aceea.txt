     1→"use client";
     2→
     3→import { useEffect, useState, useCallback, useRef, useMemo } from "react";
     4→import { io, type Socket } from "socket.io-client";
     5→import { motion, AnimatePresence } from "framer-motion";
     6→import { api, apiPost } from "@/lib/api-client";
     7→import { useUser } from "@/lib/use-user";
     8→import { navigate } from "@/lib/nav";
     9→import { Card } from "@/components/ui/card";
    10→import { Button } from "@/components/ui/button";
    11→import { Badge } from "@/components/ui/badge";
    12→import { Skeleton } from "@/components/ui/skeleton";
    13→import { Textarea } from "@/components/ui/textarea";
    14→import { EmptyState } from "@/components/shared/empty-state";
    15→import { UserAvatar } from "@/components/shared/user-avatar";
    16→import { toast } from "@/hooks/use-toast";
    17→import { timeAgoFa, toFa } from "@/lib/format";
    18→import {
    19→  MessageCircle,
    20→  Send,
    21→  ArrowRight,
    22→  Users,
    23→  Lock,
    24→  Search,
    25→  Sparkles,
    26→  ChevronLeft,
    27→  Check,
    28→  X,
    29→  Clock,
    30→  UserCheck,
    31→  Bell,
    32→} from "lucide-react";
    33→import { cn } from "@/lib/utils";
    34→
    35→/* ───────────────────────────── Types ───────────────────────────── */
    36→
    37→type OtherUser = {
    38→  id: string;
    39→  name: string;
    40→  isVerifiedBadge: boolean;
    41→  avatarUrl: string | null;
    42→  gender?: string | null;
    43→  bioShort?: string;
    44→};
    45→
    46→type ConversationListItem = {
    47→  id: string;
    48→  otherUser: OtherUser;
    49→  lastMessage: {
    50→    content: string;
    51→    createdAt: string;
    52→    senderId: string;
    53→  } | null;
    54→  status: "active" | "pending_request";
    55→  initiatorId: string | null;
    56→  unreadCount: number;
    57→  myRequestPending?: boolean;
    58→};
    59→
    60→type ChatMessage = {
    61→  id: string;
    62→  conversationId: string;
    63→  senderId: string;
    64→  content: string;
    65→  createdAt: string;
    66→};
    67→
    68→type ConversationDetail = {
    69→  conversation: {
    70→    id: string;
    71→    otherUser: OtherUser;
    72→  } | null;
    73→  messages: ChatMessage[];
    74→};
    75→
    76→/* ───────────────────────────── Main View ───────────────────────────── */
    77→
    78→export function ChatView({ conversationId }: { conversationId?: string }) {
    79→  const { user, loading: userLoading } = useUser();
    80→  const [conversations, setConversations] = useState<ConversationListItem[]>([]);
    81→  const [requests, setRequests] = useState<ConversationListItem[]>([]);
    82→  const [convLoading, setConvLoading] = useState(true);
    83→  const [activeConv, setActiveConv] = useState<ConversationDetail | null>(null);
    84→  const [msgLoading, setMsgLoading] = useState(false);
    85→  const [draft, setDraft] = useState("");
    86→  const [isTyping, setIsTyping] = useState(false);
    87→  const [search, setSearch] = useState("");
    88→  const [listTab, setListTab] = useState<"messages" | "requests">("messages");
    89→
    90→  const messagesContainerRef = useRef<HTMLDivElement | null>(null);
    91→  const socketRef = useRef<Socket | null>(null);
    92→  const typingTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    93→  const typingClearRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    94→  const activeConvRef = useRef<ConversationDetail | null>(null);
    95→  const activeConvIdRef = useRef<string | undefined>(conversationId);
    96→
    97→  /* ── Load conversation list ── */
    98→  const loadConversations = useCallback(async () => {
    99→    setConvLoading(true);
   100→    try {
   101→      const d = await api<{
   102→        conversations: ConversationListItem[];
   103→        requests: ConversationListItem[];
   104→        unreadCount: number;
   105→      }>("/api/chat/conversations");
   106→      setConversations(d.conversations || []);
   107→      setRequests(d.requests || []);
   108→    } catch (e) {
   109→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   110→    } finally {
   111→      setConvLoading(false);
   112→    }
   113→  }, []);
   114→
   115→  /* ── Load messages of active conversation ── */
   116→  const loadMessages = useCallback(async (id: string) => {
   117→    setMsgLoading(true);
   118→    try {
   119→      const d = await api<ConversationDetail>(
   120→        `/api/chat/conversations/${id}/messages`
   121→      );
   122→      setActiveConv(d);
   123→    } catch (e) {
   124→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   125→      setActiveConv(null);
   126→    } finally {
   127→      setMsgLoading(false);
   128→    }
   129→  }, []);
   130→
   131→  useEffect(() => {
   132→    if (user) loadConversations();
   133→    else setConvLoading(false);
   134→  }, [user, loadConversations]);
   135→
   136→  useEffect(() => {
   137→    activeConvIdRef.current = conversationId;
   138→    if (conversationId && user) {
   139→      loadMessages(conversationId);
   140→      // Mark messages as read
   141→      apiPost(`/api/chat/conversations/${conversationId}/read`)
   142→        .then(() => loadConversations())
   143→        .catch(() => {});
   144→    } else {
   145→      setActiveConv(null);
   146→    }
   147→  }, [conversationId, user, loadMessages, loadConversations]);
   148→
   149→  /* ── Auto-scroll helper ── */
   150→  const scrollToBottom = useCallback(() => {
   151→    const el = messagesContainerRef.current;
   152→    if (el) el.scrollTop = el.scrollHeight;
   153→  }, []);
   154→
   155→  /* ── Keep activeConv ref synced for socket handlers ── */
   156→  useEffect(() => {
   157→    activeConvRef.current = activeConv;
   158→  }, [activeConv]);
   159→
   160→  /* ── Socket.io connection (mini-service on port 3003 via gateway) ── */
   161→  useEffect(() => {
   162→    if (!user) return;
   163→    const socket = io("/", {
   164→      path: "/",
   165→      query: { XTransformPort: "3003" },
   166→      auth: { userId: user.id },
   167→      transports: ["websocket", "polling"],
   168→      reconnection: true,
   169→      reconnectionDelay: 1500,
   170→    });
   171→    socketRef.current = socket;
   172→
   173→    socket.on("connect_error", (err: Error) => {
   174→      console.warn("[chat] socket connect_error", err.message);
   175→    });
   176→
   177→    socket.on("message", (msg: ChatMessage) => {
   178→      if (activeConvRef.current?.conversation?.id === msg.conversationId) {
   179→        setActiveConv((prev) => {
   180→          if (!prev) return prev;
   181→          if (prev.messages.some((m) => m.id === msg.id)) return prev;
   182→          return { ...prev, messages: [...prev.messages, msg] };
   183→        });
   184→        queueMicrotask(() => scrollToBottom());
   185→        // Mark read if I'm the receiver
   186→        apiPost(`/api/chat/conversations/${msg.conversationId}/read`).catch(() => {});
   187→      }
   188→      // Always refresh conversation list to update last message preview + unread counts
   189→      loadConversations();
   190→    });
   191→
   192→    socket.on(
   193→      "typing",
   194→      (data: { conversationId: string; userId: string; isTyping: boolean }) => {
   195→        if (
   196→          activeConvRef.current?.conversation?.id === data.conversationId &&
   197→          data.userId !== user.id
   198→        ) {
   199→          setIsTyping(data.isTyping);
   200→          if (data.isTyping) {
   201→            if (typingClearRef.current) clearTimeout(typingClearRef.current);
   202→            typingClearRef.current = setTimeout(() => setIsTyping(false), 4000);
   203→          }
   204→        }
   205→      }
   206→    );
   207→
   208→    return () => {
   209→      socket.disconnect();
   210→      socketRef.current = null;
   211→      if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
   212→      if (typingClearRef.current) clearTimeout(typingClearRef.current);
   213→    };
   214→  }, [user, loadConversations, scrollToBottom]);
   215→
   216→  /* ── Emit join when active conversation changes ── */
   217→  useEffect(() => {
   218→    if (!conversationId || !socketRef.current) return;
   219→    if (socketRef.current.connected) {
   220→      socketRef.current.emit("join", { conversationId });
   221→    } else {
   222→      socketRef.current.once("connect", () => {
   223→        socketRef.current?.emit("join", { conversationId });
   224→      });
   225→    }
   226→  }, [conversationId]);
   227→
   228→  /* ── Auto-scroll to bottom on new messages ── */
   229→  useEffect(() => {
   230→    queueMicrotask(() => scrollToBottom());
   231→  }, [activeConv?.messages.length, scrollToBottom]);
   232→
   233→  /* ── Send a message ── */
   234→  const handleSend = useCallback(() => {
   235→    const content = draft.trim();
   236→    if (!content || !conversationId || !user || !socketRef.current) return;
   237→    setDraft("");
   238→    socketRef.current.emit("typing", {
   239→      conversationId,
   240→      userId: user.id,
   241→      isTyping: false,
   242→    });
   243→    if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
   244→
   245→    const tempId = `temp-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   246→    const optimistic: ChatMessage = {
   247→      id: tempId,
   248→      conversationId,
   249→      senderId: user.id,
   250→      content,
   251→      createdAt: new Date().toISOString(),
   252→    };
   253→    setActiveConv((prev) =>
   254→      prev ? { ...prev, messages: [...prev.messages, optimistic] } : prev
   255→    );
   256→    queueMicrotask(() => scrollToBottom());
   257→
   258→    socketRef.current.emit("message", {
   259→      conversationId,
   260→      senderId: user.id,
   261→      content,
   262→    });
   263→
   264→    setConversations((prev) =>
   265→      prev
   266→        .map((c) =>
   267→          c.id === conversationId
   268→            ? {
   269→                ...c,
   270→                lastMessage: {
   271→                  content,
   272→                  createdAt: optimistic.createdAt,
   273→                  senderId: user.id,
   274→                },
   275→              }
   276→            : c
   277→        )
   278→        .sort((a, b) => {
   279→          const at = a.lastMessage?.createdAt || "0";
   280→          const bt = b.lastMessage?.createdAt || "0";
   281→          return at < bt ? 1 : -1;
   282→        })
   283→    );
   284→  }, [draft, conversationId, user, scrollToBottom]);
   285→
   286→  const handleDraftChange = (val: string) => {
   287→    setDraft(val);
   288→    if (!conversationId || !user || !socketRef.current) return;
   289→    socketRef.current.emit("typing", {
   290→      conversationId,
   291→      userId: user.id,
   292→      isTyping: true,
   293→    });
   294→    if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
   295→    typingTimeoutRef.current = setTimeout(() => {
   296→      socketRef.current?.emit("typing", {
   297→        conversationId,
   298→        userId: user.id,
   299→        isTyping: false,
   300→      });
   301→    }, 1500);
   302→  };
   303→
   304→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   305→    if (e.key === "Enter" && !e.shiftKey) {
   306→      e.preventDefault();
   307→      handleSend();
   308→    }
   309→  };
   310→
   311→  /* ── Active conversation status (from list) ── */
   312→  const activeConvInfo = useMemo(() => {
   313→    if (!conversationId) return null;
   314→    const inList =
   315→      conversations.find((c) => c.id === conversationId) ||
   316→      requests.find((c) => c.id === conversationId);
   317→    if (inList) return inList;
   318→    // Default: assume active when not found in either list
   319→    return {
   320→      id: conversationId,
   321→      otherUser: activeConv?.conversation?.otherUser ?? {
   322→        id: "",
   323→        name: "",
   324→        isVerifiedBadge: false,
   325→        avatarUrl: null,
   326→      },
   327→      lastMessage: null,
   328→      status: "active" as const,
   329→      initiatorId: null,
   330→      unreadCount: 0,
   331→    };
   332→  }, [conversationId, conversations, requests, activeConv]);
   333→
   334→  /* ── Filtered list ── */
   335→  const filteredConversations = useMemo(() => {
   336→    if (!search.trim()) return conversations;
   337→    const q = search.trim().toLowerCase();
   338→    return conversations.filter((c) =>
   339→      c.otherUser.name.toLowerCase().includes(q)
   340→    );
   341→  }, [conversations, search]);
   342→
   343→  const filteredRequests = useMemo(() => {
   344→    if (!search.trim()) return requests;
   345→    const q = search.trim().toLowerCase();
   346→    return requests.filter((c) =>
   347→      c.otherUser.name.toLowerCase().includes(q)
   348→    );
   349→  }, [requests, search]);
   350→
   351→  /* ── Not logged in ── */
   352→  if (!userLoading && !user) {
   353→    return (
   354→      <div className="max-w-3xl mx-auto space-y-5">
   355→        <PageHeader unreadCount={0} requestsCount={0} />
   356→        <Card className="p-0 rounded-2xl border-border/60 overflow-hidden">
   357→          <EmptyState
   358→            kind="chat"
   359→            title="برای چت کردن وارد شوید"
   360→            description="برای شروع گفتگو با همکاران، ابتدا وارد حساب کاربری خود شوید."
   361→            action={
   362→              <Button
   363→                onClick={() => navigate({ view: "auth" })}
   364→                className="gap-1.5 rounded-2xl bg-primary text-primary-foreground font-bold hover:bg-primary/90"
   365→              >
   366→                <Lock className="w-4 h-4" />
   367→                ورود / ثبت‌نام
   368→              </Button>
   369→            }
   370→          />
   371→        </Card>
   372→      </div>
   373→    );
   374→  }
   375→
   376→  const showMobileList = !conversationId;
   377→
   378→  return (
   379→    <div className="space-y-4">
   380→      <PageHeader
   381→        unreadCount={conversations.reduce((s, c) => s + (c.unreadCount || 0), 0)}
   382→        requestsCount={requests.length}
   383→      />
   384→
   385→      {/* Mobile: chat list OR active chat */}
   386→      {showMobileList && (
   387→        <div className="lg:hidden">
   388→          <ChatListPanel
   389→            conversations={filteredConversations}
   390→            requests={filteredRequests}
   391→            loading={convLoading}
   392→            search={search}
   393→            onSearch={setSearch}
   394→            activeId={conversationId}
   395→            listTab={listTab}
   396→            onTabChange={setListTab}
   397→            onRespond={async (id, action) => {
   398→              try {
   399→                await apiPost(`/api/chat/conversations/${id}/respond`, { action });
   400→                toast({
   401→                  title: action === "accept" ? "درخواست تأیید شد ✅" : "درخواست رد شد",
   402→                });
   403→                await loadConversations();
   404→              } catch (e) {
   405→                toast({
   406→                  title: "خطا",
   407→                  description: (e as Error).message,
   408→                  variant: "destructive",
   409→                });
   410→              }
   411→            }}
   412→          />
   413→        </div>
   414→      )}
   415→
   416→      {!showMobileList && (
   417→        <div className="lg:hidden">
   418→          <ChatThread
   419→            conv={activeConv}
   420→            convInfo={activeConvInfo}
   421→            loading={msgLoading}
   422→            draft={draft}
   423→            onDraftChange={handleDraftChange}
   424→            onKeyDown={handleKeyDown}
   425→            onSend={handleSend}
   426→            currentUserId={user?.id || ""}
   427→            isTyping={isTyping}
   428→            messagesContainerRef={messagesContainerRef}
   429→            onBack={() => navigate({ view: "chat" })}
   430→            onRespond={async (id, action) => {
   431→              try {
   432→                await apiPost(`/api/chat/conversations/${id}/respond`, { action });
   433→                toast({
   434→                  title: action === "accept" ? "درخواست تأیید شد ✅" : "درخواست رد شد",
   435→                });
   436→                if (action === "accept") {
   437→                  await loadMessages(id);
   438→                  await loadConversations();
   439→                } else {
   440→                  navigate({ view: "chat" });
   441→                }
   442→              } catch (e) {
   443→                toast({
   444→                  title: "خطا",
   445→                  description: (e as Error).message,
   446→                  variant: "destructive",
   447→                });
   448→              }
   449→            }}
   450→          />
   451→        </div>
   452→      )}
   453→
   454→      {/* Desktop: two-pane — list on right (RTL start), chat on left */}
   455→      <div className="hidden lg:grid lg:grid-cols-[1fr_340px] gap-4">
   456→        <div>
   457→          {conversationId ? (
   458→            <ChatThread
   459→              conv={activeConv}
   460→              convInfo={activeConvInfo}
   461→              loading={msgLoading}
   462→              draft={draft}
   463→              onDraftChange={handleDraftChange}
   464→              onKeyDown={handleKeyDown}
   465→              onSend={handleSend}
   466→              currentUserId={user?.id || ""}
   467→              isTyping={isTyping}
   468→              messagesContainerRef={messagesContainerRef}
   469→              onRespond={async (id, action) => {
   470→                try {
   471→                  await apiPost(`/api/chat/conversations/${id}/respond`, { action });
   472→                  toast({
   473→                    title: action === "accept" ? "درخواست تأیید شد ✅" : "درخواست رد شد",
   474→                  });
   475→                  if (action === "accept") {
   476→                    await loadMessages(id);
   477→                    await loadConversations();
   478→                  } else {
   479→                    navigate({ view: "chat" });
   480→                  }
   481→                } catch (e) {
   482→                  toast({
   483→                    title: "خطا",
   484→                    description: (e as Error).message,
   485→                    variant: "destructive",
   486→                  });
   487→                }
   488→              }}
   489→            />
   490→          ) : (
   491→            <Card className="h-[72vh] flex items-center justify-center rounded-2xl border-border/60 bg-muted/30">
   492→              <EmptyState
   493→                kind="chat"
   494→                title="یک گفتگو را انتخاب کنید"
   495→                description="از لیست کناری یک گفتگو را انتخاب کنید یا همکار جدیدی پیدا کنید."
   496→                action={
   497→                  <Button
   498→                    variant="outline"
   499→                    onClick={() => navigate({ view: "talents" })}
   500→                    className="gap-1.5 rounded-2xl border-primary/30 text-primary hover:bg-primary/5"
   501→                  >
   502→                    <Users className="w-4 h-4" />
   503→                    پیدا کردن همکار
   504→                  </Button>
   505→                }
   506→              />
   507→            </Card>
   508→          )}
   509→        </div>
   510→        <div>
   511→          <ChatListPanel
   512→            conversations={filteredConversations}
   513→            requests={filteredRequests}
   514→            loading={convLoading}
   515→            search={search}
   516→            onSearch={setSearch}
   517→            activeId={conversationId}
   518→            listTab={listTab}
   519→            onTabChange={setListTab}
   520→            onRespond={async (id, action) => {
   521→              try {
   522→                await apiPost(`/api/chat/conversations/${id}/respond`, { action });
   523→                toast({
   524→                  title: action === "accept" ? "درخواست تأیید شد ✅" : "درخواست رد شد",
   525→                });
   526→                await loadConversations();
   527→                if (action === "accept") {
   528→                  navigate({ view: "chat", conversationId: id });
   529→                }
   530→              } catch (e) {
   531→                toast({
   532→                  title: "خطا",
   533→                  description: (e as Error).message,
   534→                  variant: "destructive",
   535→                });
   536→              }
   537→            }}
   538→          />
   539→        </div>
   540→      </div>
   541→    </div>
   542→  );
   543→}
   544→
   545→/* ───────────────────────────── Page Header ───────────────────────────── */
   546→
   547→function PageHeader({
   548→  unreadCount,
   549→  requestsCount,
   550→}: {
   551→  unreadCount: number;
   552→  requestsCount: number;
   553→}) {
   554→  return (
   555→    <motion.div
   556→      initial={{ opacity: 0, y: -8 }}
   557→      animate={{ opacity: 1, y: 0 }}
   558→      className="flex items-center justify-between"
   559→    >
   560→      <div className="flex items-center gap-3">
   561→        <div className="grid place-items-center w-11 h-11 rounded-2xl bg-primary text-primary-foreground shadow-md">
   562→          <MessageCircle className="w-5 h-5" />
   563→        </div>
   564→        <div>
   565→          <h1 className="text-xl font-extrabold leading-tight tracking-tight">چت</h1>
   566→          <p className="text-xs text-muted-foreground mt-0.5">
   567→            گفتگوی زنده با همکاران
   568→          </p>
   569→        </div>
   570→      </div>
   571→      <div className="flex items-center gap-2">
   572→        {requestsCount > 0 && (
   573→          <Badge className="bg-warning/15 text-warning border border-warning/30 rounded-full font-bold">
   574→            {toFa(requestsCount)} درخواست
   575→          </Badge>
   576→        )}
   577→        {unreadCount > 0 && (
   578→          <Badge className="bg-primary/15 text-primary border border-primary/30 rounded-full font-bold">
   579→            {toFa(unreadCount)} پیام جدید
   580→          </Badge>
   581→        )}
   582→        <Button
   583→          variant="outline"
   584→          size="sm"
   585→          onClick={() => navigate({ view: "talents" })}
   586→          className="gap-1.5 rounded-2xl border-primary/30 text-primary hover:bg-primary/5"
   587→        >
   588→          <Users className="w-4 h-4" />
   589→          <span className="hidden sm:inline">همکاران</span>
   590→        </Button>
   591→      </div>
   592→    </motion.div>
   593→  );
   594→}
   595→
   596→/* ───────────────────────────── Chat List Panel ───────────────────────────── */
   597→
   598→function ChatListPanel({
   599→  conversations,
   600→  requests,
   601→  loading,
   602→  search,
   603→  onSearch,
   604→  activeId,
   605→  listTab,
   606→  onTabChange,
