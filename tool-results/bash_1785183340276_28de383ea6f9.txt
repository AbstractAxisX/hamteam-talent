    62→function hashIdToIndex(id: string, mod: number): number {
    63→  let h = 0;
    64→  for (let i = 0; i < id.length; i++) {
    65→    h = (h * 31 + id.charCodeAt(i)) >>> 0;
    66→  }
    67→  return h % mod;
    68→}

   322→// ─── Section 2: Location ───────────────────────────────────────────
   323→function LocationSection({
   324→  profile,
   325→  onUpdated,
   326→}: {
   327→  profile: ProfileDetail;
   328→  onUpdated: () => void;
   329→}) {
   330→  const [province, setProvince] = useState(profile.province ?? "");
   331→  const [city, setCity] = useState(profile.city ?? "");
   332→  const [phoneVisible, setPhoneVisible] = useState(profile.phoneVisible);
   333→  const [saving, setSaving] = useState(false);
   334→
   335→  useEffect(() => {
   336→    setProvince(profile.province ?? "");
   337→    setCity(profile.city ?? "");
   338→    setPhoneVisible(profile.phoneVisible);
   339→  }, [profile]);
   340→
   341→  const provinceObj = PROVINCES.find((p) => p.id === province);
   342→  const cities = provinceObj?.cities ?? [];
   343→
   344→  async function save() {
   345→    setSaving(true);
   346→    try {
   347→      await apiPut("/api/profile/me", {
   348→        province: province || null,
   349→        city: city || null,
   350→        phoneVisible,
   351→      });
   352→      toast({ title: "ذخیره شد ✅" });
   353→      onUpdated();
   354→    } catch (e) {
   355→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   356→    } finally {
   357→      setSaving(false);
   358→    }
   359→  }
   360→
   361→  return (
   362→    <Card className="p-5 md:p-6 space-y-4">
   363→      <SectionTitle icon={MapPin} title="موقعیت و تماس" />
   364→
   365→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   366→        <div className="space-y-2">
   367→          <Label>استان</Label>
   368→          <Select
   369→            value={province}
   370→            onValueChange={(v) => {
   371→              setProvince(v);
   372→              setCity("");
   373→            }}
   374→          >
   375→            <SelectTrigger className="w-full">
   376→              <SelectValue placeholder="انتخاب استان" />
   377→            </SelectTrigger>
   378→            <SelectContent>
   379→              {PROVINCES.map((p) => (
   380→                <SelectItem key={p.id} value={p.id}>
   381→                  {p.name}
   382→                </SelectItem>
   383→              ))}
   384→            </SelectContent>
   385→          </Select>
   386→        </div>
   387→        <div className="space-y-2">
   388→          <Label>شهر</Label>
   389→          <Select value={city} onValueChange={setCity} disabled={!province}>
   390→            <SelectTrigger className="w-full">
   391→              <SelectValue placeholder="انتخاب شهر" />
   392→            </SelectTrigger>
   393→            <SelectContent>
   394→              {cities.map((c) => (
   395→                <SelectItem key={c} value={c}>
   396→                  {c}
   397→                </SelectItem>
   398→              ))}
   399→            </SelectContent>
   400→          </Select>
   401→        </div>
   402→      </div>
   403→
   404→      <div className="flex items-start justify-between gap-3 rounded-lg bg-muted/40 p-3">
   405→        <div className="flex items-start gap-2">
   406→          <Phone className="w-4 h-4 text-primary mt-0.5" />
   407→          <div>
   408→            <p className="text-sm font-medium">نمایش شماره تلفن</p>
   409→            <p className="text-xs text-muted-foreground">
   410→              {phoneVisible
   411→                ? `شماره ${toFa(profile.phone || "")} برای بازدیدکنندگان قابل مشاهده است.`
   412→                : "شماره تلفن شما در پروفایل عمومی پنهان است."}
   413→            </p>
   414→          </div>
   415→        </div>
   416→        <Switch checked={phoneVisible} onCheckedChange={setPhoneVisible} />
   417→      </div>
   418→
   419→      <div className="flex justify-end">
   420→        <Button onClick={save} disabled={saving} className="gap-1.5">
   421→          {saving ? <Loader2 className="w-4 h-4 animate-spin" /> : <Save className="w-4 h-4" />}
   422→          ذخیره
   423→        </Button>
   424→      </div>
   425→    </Card>
   426→  );

   429→// ─── Section 3: Categories & Skills ───────────────────────────────
   430→function CategoriesSection({
   431→  profile,
   432→  allCats,
   433→  onUpdated,
   434→}: {
   435→  profile: ProfileDetail;
   436→  allCats: CategoryWithSkills[];
   437→  onUpdated: () => void;
   438→}) {
   439→  const [addCatOpen, setAddCatOpen] = useState(false);
   440→  const [addSkillCatId, setAddSkillCatId] = useState<string | null>(null);
   441→  const [busy, setBusy] = useState(false);
   442→
   443→  const selectedCatIds = new Set(profile.categories.map((c) => c.id));
   444→  const availableCats = allCats.filter((c) => !selectedCatIds.has(c.id));
   445→
   446→  async function addCategory(categoryId: string) {
   447→    setBusy(true);
   448→    try {
   449→      await apiPost("/api/profile/me/categories", { categoryId });
   450→      toast({ title: "دسته‌بندی اضافه شد ✅" });
   451→      setAddCatOpen(false);
   452→      onUpdated();
   453→    } catch (e) {
   454→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   455→    } finally {
   456→      setBusy(false);
   457→    }
   458→  }
   459→
   460→  async function removeCategory(categoryId: string) {
   461→    setBusy(true);
   462→    try {
   463→      await apiDelete(`/api/profile/me/categories/${categoryId}`);
   464→      toast({ title: "حذف شد" });
   465→      onUpdated();
   466→    } catch (e) {
   467→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   468→    } finally {
   469→      setBusy(false);
   470→    }
   471→  }
   472→
   473→  async function addSkill(skillId: string) {
   474→    if (!addSkillCatId) return;
   475→    setBusy(true);
   476→    try {
   477→      await apiPost("/api/profile/me/skills", { skillId });
   478→      toast({ title: "مهارت اضافه شد ✅" });
   479→      setAddSkillCatId(null);
   480→      onUpdated();
   481→    } catch (e) {
   482→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   483→    } finally {
   484→      setBusy(false);
   485→    }
   486→  }
   487→
   488→  async function removeSkill(skillId: string) {
   489→    setBusy(true);
   490→    try {
   491→      await apiDelete(`/api/profile/me/skills/${skillId}`);
   492→      toast({ title: "حذف شد" });
   493→      onUpdated();
   494→    } catch (e) {
   495→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   496→    } finally {
   497→      setBusy(false);
   498→    }
   499→  }
   500→
   501→  // Compute available skills for "add skill" dialog
   502→  const addSkillCat = profile.categories.find((c) => c.id === addSkillCatId);
   503→  const addSkillCatFull = allCats.find((c) => c.id === addSkillCatId);
   504→  const selectedSkillIds = new Set(addSkillCat?.skills.map((s) => s.id) ?? []);
   505→  const availableSkills = (addSkillCatFull?.skills ?? []).filter(
   506→    (s) => !selectedSkillIds.has(s.id)
   507→  );
   508→
   509→  return (
   510→    <Card className="p-5 md:p-6 space-y-4">
   511→      <div className="flex items-center justify-between gap-2">
   512→        <SectionTitle icon={Hash} title="دسته‌بندی و مهارت‌ها" />
   513→        <Dialog open={addCatOpen} onOpenChange={setAddCatOpen}>
   514→          <DialogTrigger asChild>
   515→            <Button size="sm" variant="outline" className="gap-1.5" disabled={availableCats.length === 0}>
   516→              <Plus className="w-4 h-4" /> افزودن دسته‌بندی
   517→            </Button>
   518→          </DialogTrigger>
   519→          <DialogContent>
   520→            <DialogHeader>
   521→              <DialogTitle>افزودن دسته‌بندی</DialogTitle>
   522→              <DialogDescription>
   523→                یک دسته‌بندی تخصصی به پروفایل خود اضافه کنید.
   524→              </DialogDescription>
   525→            </DialogHeader>
   526→            <div className="max-h-72 overflow-y-auto slim-scroll space-y-1 pr-1">
   527→              {availableCats.map((c) => (
   528→                <button
   529→                  key={c.id}
   530→                  onClick={() => addCategory(c.id)}
   531→                  disabled={busy}
   532→                  className="w-full text-right px-3 py-2 rounded-md hover:bg-accent text-sm flex items-center justify-between"
   533→                >
   534→                  <span>{c.name}</span>
   535→                  <Plus className="w-4 h-4 text-primary" />
   536→                </button>
   537→              ))}
   538→            </div>
   539→          </DialogContent>
   540→        </Dialog>
   541→      </div>
   542→
   543→      {profile.categories.length === 0 ? (
   544→        <EmptyState
   545→          icon={Hash}
   546→          title="هنوز تخصصی ثبت نشده"
   547→          description="برای نمایش در پروفایل و انتشار پست، حداقل یک دسته‌بندی و مهارت اضافه کنید."
