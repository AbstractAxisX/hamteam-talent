    62→    62→function hashIdToIndex(id: string, mod: number): number {
    63→    63→  let h = 0;
    64→    64→  for (let i = 0; i < id.length; i++) {
    65→    65→    h = (h * 31 + id.charCodeAt(i)) >>> 0;
    66→    66→  }
    67→    67→  return h % mod;
    68→    68→}
