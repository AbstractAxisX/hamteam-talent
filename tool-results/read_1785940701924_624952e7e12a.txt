    63→    63→    63→
    64→    64→    64→export function ChatView({ conversationId }: { conversationId?: string }) {
    65→    65→    65→  const { user, loading: userLoading } = useUser();
    66→    66→    66→  const [conversations, setConversations] = useState<ConversationListItem[]>([]);
    67→    67→    67→  const [convLoading, setConvLoading] = useState(true);
    68→    68→    68→  const [activeConv, setActiveConv] = useState<ConversationDetail | null>(null);
    69→    69→    69→  const [msgLoading, setMsgLoading] = useState(false);
    70→    70→    70→  const [draft, setDraft] = useState("");
    71→    71→    71→  const [isTyping, setIsTyping] = useState(false);
    72→    72→    72→  const [search, setSearch] = useState("");
    73→    73→    73→
    74→    74→    74→  const messagesContainerRef = useRef<HTMLDivElement | null>(null);
    75→    75→    75→  const socketRef = useRef<Socket | null>(null);
    76→    76→    76→  const typingTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    77→    77→    77→  const typingClearRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    78→    78→    78→  const activeConvRef = useRef<ConversationDetail | null>(null);
    79→    79→    79→
    80→    80→    80→  /* ── Load conversation list ── */
    81→    81→    81→  const loadConversations = useCallback(async () => {
    82→    82→    82→    setConvLoading(true);
    83→    83→    83→    try {
    84→    84→    84→      const d = await api<{ conversations: ConversationListItem[] }>(
    85→    85→    85→        "/api/chat/conversations"
    86→    86→    86→      );
    87→    87→    87→      setConversations(d.conversations);
    88→    88→    88→    } catch (e) {
    89→    89→    89→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
    90→    90→    90→    } finally {
    91→    91→    91→      setConvLoading(false);
    92→    92→    92→    }
    93→    93→    93→  }, []);
    94→    94→    94→
    95→    95→    95→  /* ── Load messages of active conversation ── */
    96→    96→    96→  const loadMessages = useCallback(async (id: string) => {
    97→    97→    97→    setMsgLoading(true);
    98→    98→    98→    try {
    99→    99→    99→      const d = await api<ConversationDetail>(
   100→   100→   100→        `/api/chat/conversations/${id}/messages`
   101→   101→   101→      );
   102→   102→   102→      setActiveConv(d);
   103→   103→   103→    } catch (e) {
   104→   104→   104→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   105→   105→   105→      setActiveConv(null);
   106→   106→   106→    } finally {
   107→   107→   107→      setMsgLoading(false);
   108→   108→   108→    }
   109→   109→   109→  }, []);
   110→   110→   110→
   111→   111→   111→  useEffect(() => {
   112→   112→   112→    if (user) loadConversations();
   113→   113→   113→    else setConvLoading(false);
   114→   114→   114→  }, [user, loadConversations]);
   115→   115→   115→
   116→   116→   116→  useEffect(() => {
   117→   117→   117→    if (conversationId && user) loadMessages(conversationId);
   118→   118→   118→    else setActiveConv(null);
   119→   119→   119→  }, [conversationId, user, loadMessages]);
   120→   120→   120→
   121→   121→   121→  /* ── Auto-scroll helper ── */
   122→   122→   122→  const scrollToBottom = useCallback(() => {
   123→   123→   123→    const el = messagesContainerRef.current;
   124→   124→   124→    if (el) el.scrollTop = el.scrollHeight;
   125→   125→   125→  }, []);
   126→   126→   126→
   127→   127→   127→  /* ── Keep activeConv ref synced for socket handlers ── */
   128→   128→   128→  useEffect(() => {
   129→   129→   129→    activeConvRef.current = activeConv;
   130→   130→   130→  }, [activeConv]);
   131→   131→   131→
   132→   132→   132→  /* ── Socket.io connection ── */
   133→   133→   133→  useEffect(() => {
   134→   134→   134→    if (!user) return;
   135→   135→   135→    const socket = io("/", {
   136→   136→   136→      path: "/",
   137→   137→   137→      query: { XTransformPort: "3003" },
   138→   138→   138→      auth: { userId: user.id },
   139→   139→   139→      transports: ["websocket", "polling"],
   140→   140→   140→      reconnection: true,
   141→   141→   141→      reconnectionDelay: 1500,
   142→   142→   142→    });
   143→   143→   143→    socketRef.current = socket;
   144→   144→   144→
   145→   145→   145→    socket.on("connect_error", (err: Error) => {
   146→   146→   146→      console.warn("[chat] socket connect_error", err.message);
   147→   147→   147→    });
   148→   148→   148→
   149→   149→   149→    socket.on("message", (msg: ChatMessage) => {
   150→   150→   150→      if (activeConvRef.current?.conversation?.id === msg.conversationId) {
   151→   151→   151→        setActiveConv((prev) => {
   152→   152→   152→          if (!prev) return prev;
   153→   153→   153→          if (prev.messages.some((m) => m.id === msg.id)) return prev;
   154→   154→   154→          return { ...prev, messages: [...prev.messages, msg] };
   155→   155→   155→        });
   156→   156→   156→        queueMicrotask(() => scrollToBottom());
   157→   157→   157→      }
   158→   158→   158→      // Always refresh conversation list to update last message preview
   159→   159→   159→      loadConversations();
   160→   160→   160→    });
   161→   161→   161→
   162→   162→   162→    socket.on(
   163→   163→   163→      "typing",
   164→   164→   164→      (data: { conversationId: string; userId: string; isTyping: boolean }) => {
   165→   165→   165→        if (
   166→   166→   166→          activeConvRef.current?.conversation?.id === data.conversationId &&
   167→   167→   167→          data.userId !== user.id
   168→   168→   168→        ) {
   169→   169→   169→          setIsTyping(data.isTyping);
   170→   170→   170→          if (data.isTyping) {
   171→   171→   171→            if (typingClearRef.current) clearTimeout(typingClearRef.current);
   172→   172→   172→            typingClearRef.current = setTimeout(() => setIsTyping(false), 4000);
   173→   173→   173→          }
   174→   174→   174→        }
   175→   175→   175→      }
   176→   176→   176→    );
   177→   177→   177→
   178→   178→   178→    return () => {
   179→   179→   179→      socket.disconnect();
   180→   180→   180→      socketRef.current = null;
   181→   181→   181→      if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
   182→   182→   182→      if (typingClearRef.current) clearTimeout(typingClearRef.current);
   183→   183→   183→    };
   184→   184→   184→  }, [user, loadConversations, scrollToBottom]);
   185→   185→   185→
   186→   186→   186→  /* ── Emit join when active conversation changes ── */
   187→   187→   187→  useEffect(() => {
   188→   188→   188→    if (!conversationId || !socketRef.current) return;
   189→   189→   189→    if (socketRef.current.connected) {
   190→   190→   190→      socketRef.current.emit("join", { conversationId });
   191→   191→   191→    } else {
   192→   192→   192→      socketRef.current.once("connect", () => {
   193→   193→   193→        socketRef.current?.emit("join", { conversationId });
   194→   194→   194→      });
   195→   195→   195→    }
   196→   196→   196→  }, [conversationId]);
   197→   197→   197→
   198→   198→   198→  /* ── Auto-scroll to bottom on new messages ── */
   199→   199→   199→  useEffect(() => {
   200→   200→   200→    queueMicrotask(() => scrollToBottom());
   201→   201→   201→  }, [activeConv?.messages.length, scrollToBottom]);
   202→   202→   202→
   203→   203→   203→  /* ── Send a message ── */
   204→   204→   204→  const handleSend = useCallback(async () => {
   205→   205→   205→    const content = draft.trim();
   206→   206→   206→    if (!content || !conversationId || !user || !socketRef.current) return;
   207→   207→   207→    setDraft("");
   208→   208→   208→    socketRef.current.emit("typing", {
   209→   209→   209→      conversationId,
   210→   210→   210→      userId: user.id,
   211→   211→   211→      isTyping: false,
   212→   212→   212→    });
   213→   213→   213→    if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
   214→   214→   214→
   215→   215→   215→    const tempId = `temp-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
   216→   216→   216→    const optimistic: ChatMessage = {
   217→   217→   217→      id: tempId,
   218→   218→   218→      conversationId,
   219→   219→   219→      senderId: user.id,
   220→   220→   220→      content,
   221→   221→   221→      createdAt: new Date().toISOString(),
   222→   222→   222→    };
   223→   223→   223→    setActiveConv((prev) =>
   224→   224→   224→      prev ? { ...prev, messages: [...prev.messages, optimistic] } : prev
   225→   225→   225→    );
   226→   226→   226→    queueMicrotask(() => scrollToBottom());
   227→   227→   227→
   228→   228→   228→    socketRef.current.emit("message", {
   229→   229→   229→      conversationId,
   230→   230→   230→      senderId: user.id,
   231→   231→   231→      content,
   232→   232→   232→    });
   233→   233→   233→
   234→   234→   234→    setConversations((prev) =>
   235→   235→   235→      prev
   236→   236→   236→        .map((c) =>
   237→   237→   237→          c.id === conversationId
   238→   238→   238→            ? {
   239→   239→   239→                ...c,
   240→   240→   240→                lastMessage: {
   241→   241→   241→                  content,
   242→   242→   242→                  createdAt: optimistic.createdAt,
   243→   243→   243→                  senderId: user.id,
   244→   244→   244→                },
   245→   245→   245→              }
   246→   246→   246→            : c
   247→   247→   247→        )
   248→   248→   248→        .sort((a, b) => {
   249→   249→   249→          const at = a.lastMessage?.createdAt || "0";
   250→   250→   250→          const bt = b.lastMessage?.createdAt || "0";
   251→   251→   251→          return at < bt ? 1 : -1;
   252→   252→   252→        })
   253→   253→   253→    );
   254→   254→   254→  }, [draft, conversationId, user, scrollToBottom]);
   255→   255→   255→
   256→   256→   256→  const handleDraftChange = (val: string) => {
   257→   257→   257→    setDraft(val);
   258→   258→   258→    if (!conversationId || !user || !socketRef.current) return;
   259→   259→   259→    socketRef.current.emit("typing", {
   260→   260→   260→      conversationId,
   261→   261→   261→      userId: user.id,
   262→   262→   262→      isTyping: true,
   263→   263→   263→    });
   264→   264→   264→    if (typingTimeoutRef.current) clearTimeout(typingTimeoutRef.current);
   265→   265→   265→    typingTimeoutRef.current = setTimeout(() => {
   266→   266→   266→      socketRef.current?.emit("typing", {
   267→   267→   267→        conversationId,
   268→   268→   268→        userId: user.id,
   269→   269→   269→        isTyping: false,
   270→   270→   270→      });
   271→   271→   271→    }, 1500);
   272→   272→   272→  };
   273→   273→   273→
   274→   274→   274→  const handleKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   275→   275→   275→    if (e.key === "Enter" && !e.shiftKey) {
   276→   276→   276→      e.preventDefault();
   277→   277→   277→      handleSend();
   278→   278→   278→    }
   279→   279→   279→  };
   280→   280→   280→
   281→   281→   281→  /* ── Filtered list ── */
   282→   282→   282→  const filteredConversations = useMemo(() => {
   283→   283→   283→    if (!search.trim()) return conversations;
   284→   284→   284→    const q = search.trim().toLowerCase();
   285→   285→   285→    return conversations.filter((c) =>
   286→   286→   286→      c.otherUser.name.toLowerCase().includes(q)
   287→   287→   287→    );
   288→   288→   288→  }, [conversations, search]);
   289→   289→   289→
   290→   290→   290→  /* ── Not logged in ── */
   291→   291→   291→  if (!userLoading && !user) {
   292→   292→   292→    return (
   293→   293→   293→      <div className="max-w-3xl mx-auto space-y-5">
   294→   294→   294→        <PageHeader
   295→   295→   295→          icon={MessageCircle}
   296→   296→   296→          title="چت"
   297→   297→   297→          subtitle="گفتگوی زنده با همکاران"
   298→   298→   298→        />
   299→   299→   299→        <Card className="p-0 rounded-2xl border-border/60 shadow-card overflow-hidden">
   300→   300→   300→          <EmptyState
   301→   301→   301→            kind="chat"
   302→   302→   302→            title="برای چت کردن وارد شوید"
   303→   303→   303→            description="برای شروع گفتگو با همکاران، ابتدا وارد حساب کاربری خود شوید."
   304→   304→   304→            action={
   305→   305→   305→              <Button
   306→   306→   306→                onClick={() => navigate({ view: "auth" })}
   307→   307→   307→                className="gap-1.5 rounded-xl"
   308→   308→   308→              >
   309→   309→   309→                <Lock className="w-4 h-4" />
   310→   310→   310→                ورود / ثبت‌نام
   311→   311→   311→              </Button>
   312→   312→   312→            }
   313→   313→   313→          />
   314→   314→   314→        </Card>
   315→   315→   315→      </div>
   316→   316→   316→    );
   317→   317→   317→  }
   318→   318→   318→
   319→   319→   319→  const showMobileList = !conversationId;
   320→   320→   320→
   321→   321→   321→  return (
   322→   322→   322→    <div className="space-y-4">
   323→   323→   323→      <PageHeader
   324→   324→   324→        icon={MessageCircle}
   325→   325→   325→        title="چت"
   326→   326→   326→        subtitle="گفتگوی زنده با همکاران"
   327→   327→   327→      />
   328→   328→   328→
   329→   329→   329→      {/* Mobile: chat list */}
   330→   330→   330→      {showMobileList && (
   331→   331→   331→        <div className="lg:hidden">
   332→   332→   332→          <ChatList
   333→   333→   333→            conversations={filteredConversations}
   334→   334→   334→            loading={convLoading}
   335→   335→   335→            search={search}
   336→   336→   336→            onSearch={setSearch}
   337→   337→   337→            activeId={conversationId}
   338→   338→   338→          />
   339→   339→   339→        </div>
   340→   340→   340→      )}
   341→   341→   341→
   342→   342→   342→      {/* Mobile: active chat */}
   343→   343→   343→      {!showMobileList && (
   344→   344→   344→        <div className="lg:hidden">
   345→   345→   345→          <ChatThread
   346→   346→   346→            conv={activeConv}
   347→   347→   347→            loading={msgLoading}
   348→   348→   348→            draft={draft}
   349→   349→   349→            onDraftChange={handleDraftChange}
   350→   350→   350→            onKeyDown={handleKeyDown}
   351→   351→   351→            onSend={handleSend}
   352→   352→   352→            currentUserId={user?.id || ""}
   353→   353→   353→            isTyping={isTyping}
   354→   354→   354→            messagesContainerRef={messagesContainerRef}
   355→   355→   355→            onBack={() => navigate({ view: "chat" })}
   356→   356→   356→          />
   357→   357→   357→        </div>
   358→   358→   358→      )}
   359→   359→   359→
   360→   360→   360→      {/* Desktop: two-pane — list on right (RTL start), chat on left */}
   361→   361→   361→      <div className="hidden lg:grid lg:grid-cols-[320px_1fr] gap-4">
   362→   362→   362→        <div>
   363→   363→   363→          <ChatList
   364→   364→   364→            conversations={filteredConversations}
   365→   365→   365→            loading={convLoading}
   366→   366→   366→            search={search}
   367→   367→   367→            onSearch={setSearch}
   368→   368→   368→            activeId={conversationId}
   369→   369→   369→          />
   370→   370→   370→        </div>
   371→   371→   371→        <div>
   372→   372→   372→          {conversationId ? (
   373→   373→   373→            <ChatThread
   374→   374→   374→              conv={activeConv}
   375→   375→   375→              loading={msgLoading}
   376→   376→   376→              draft={draft}
   377→   377→   377→              onDraftChange={handleDraftChange}
   378→   378→   378→              onKeyDown={handleKeyDown}
   379→   379→   379→              onSend={handleSend}
   380→   380→   380→              currentUserId={user?.id || ""}
   381→   381→   381→              isTyping={isTyping}
   382→   382→   382→              messagesContainerRef={messagesContainerRef}
   383→   383→   383→            />
   384→   384→   384→          ) : (
   385→   385→   385→            <Card className="h-[70vh] flex items-center justify-center rounded-2xl border-border/60 shadow-card">
   386→   386→   386→              <EmptyState
   387→   387→   387→                kind="chat"
   388→   388→   388→                title="یک گفتگو را انتخاب کنید"
   389→   389→   389→                description="برای شروع گفتگوی جدید، از پروفایل همکاران یا صفحه‌ی افراد استفاده کنید."
   390→   390→   390→                action={
   391→   391→   391→                  <Button
   392→   392→   392→                    variant="outline"
   393→   393→   393→                    onClick={() => navigate({ view: "people" })}
   394→   394→   394→                    className="gap-1.5 rounded-xl"
   395→   395→   395→                  >
   396→   396→   396→                    <Users className="w-4 h-4" />
   397→   397→   397→                    پیدا کردن همکار
   398→   398→   398→                  </Button>
   399→   399→   399→                }
   400→   400→   400→              />
   401→   401→   401→            </Card>
   402→   402→   402→          )}
   403→   403→   403→        </div>
   404→   404→   404→      </div>
   405→   405→   405→    </div>
   406→   406→   406→  );
   407→   407→   407→}

   410→   410→   410→
   411→   411→   411→function PageHeader({
   412→   412→   412→  icon: Icon,
   413→   413→   413→  title,
   414→   414→   414→  subtitle,
   415→   415→   415→}: {
   416→   416→   416→  icon: React.ComponentType<{ className?: string }>;
   417→   417→   417→  title: string;
   418→   418→   418→  subtitle: string;
   419→   419→   419→}) {
   420→   420→   420→  return (
   421→   421→   421→    <div className="flex items-center gap-3">
   422→   422→   422→      <div className="grid place-items-center w-11 h-11 rounded-2xl bg-brand-gradient text-white shadow-card">
   423→   423→   423→        <Icon className="w-5 h-5" />
