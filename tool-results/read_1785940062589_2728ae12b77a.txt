    53→function hashIdToIndex(id: string, mod: number): number {
    54→  let h = 0;
    55→  for (let i = 0; i < id.length; i++) {
    56→    h = (h * 31 + id.charCodeAt(i)) >>> 0;
    57→  }
    58→  return h % mod;
    59→}

   366→function ConnectionButton({
   367→  profile,
   368→  onUpdated,
   369→}: {
   370→  profile: ProfileDetail;
   371→  onUpdated: () => void;
   372→}) {
   373→  const [busy, setBusy] = useState(false);
   374→  const status = profile.connectionStatus;
   375→
   376→  async function handle() {
   377→    if (status === "accepted") {
   378→      toast({ title: "شما قبلاً متصل شده‌اید" });
   379→      return;
   380→    }
   381→    setBusy(true);
   382→    try {
   383→      const res = await apiPost<{ status: string }>("/api/connections", {
   384→        receiverId: profile.userId,
   385→      });
   386→      if (res.status === "accepted") {
   387→        toast({ title: "ارتباط برقرار شد ✅" });
   388→      } else if (res.status === "pending-sent") {
   389→        toast({ title: "درخواست ارتباط ارسال شد 📨" });
   390→      }
   391→      onUpdated();
   392→    } catch (e) {
   393→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   394→    } finally {
   395→      setBusy(false);
   396→    }
   397→  }
   398→
   399→  if (status === "accepted") {
   400→    return (
   401→      <Button variant="outline" onClick={handle} disabled className="gap-1.5 rounded-xl font-semibold">
   402→        <UserCheck className="w-4 h-4 text-primary" /> متصل
   403→      </Button>
   404→    );
   405→  }
   406→  if (status === "pending-sent") {
   407→    return (
   408→      <Button variant="outline" onClick={handle} disabled className="gap-1.5 rounded-xl font-semibold">
   409→        <Clock className="w-4 h-4" /> در انتظار
   410→      </Button>
   411→    );
   412→  }
   413→  if (status === "pending-received") {
   414→    return (
   415→      <Button onClick={handle} disabled={busy} className="gap-1.5 rounded-xl font-semibold">
   416→        {busy ? <Loader2 className="w-4 h-4 animate-spin" /> : <UserCheck className="w-4 h-4" />}
   417→        پذیرش درخواست
   418→      </Button>
   419→    );
   420→  }
   421→  return (
   422→    <Button onClick={handle} disabled={busy} className="gap-1.5 rounded-xl font-semibold">
   423→      {busy ? <Loader2 className="w-4 h-4 animate-spin" /> : <UserPlus className="w-4 h-4" />}
   424→      دنبال کردن
   425→    </Button>
   426→  );
   427→}

   662→function PostsTab({ userId, isSelf }: { userId: string; isSelf: boolean }) {
   663→  const [posts, setPosts] = useState<PostWithRelations[] | null>(null);
   664→  const [loading, setLoading] = useState(true);
   665→
   666→  const load = useCallback(async () => {
   667→    setLoading(true);
   668→    try {
   669→      const data = await api<{ posts: PostWithRelations[] }>(`/api/posts?userId=${userId}`);
   670→      setPosts(data.posts);
   671→    } catch (e) {
   672→      toast({ title: "خطا", description: (e as Error).message, variant: "destructive" });
   673→      setPosts([]);
   674→    } finally {
   675→      setLoading(false);
   676→    }
   677→  }, [userId]);
   678→
   679→  useEffect(() => {
   680→    load();
   681→  }, [load]);
   682→
   683→  if (loading) {
   684→    return (
   685→      <div className="space-y-3">
   686→        {[...Array(2)].map((_, i) => (
   687→          <Card key={i} className="p-4 space-y-3 border-border/60 shadow-card">
   688→            <div className="flex items-center gap-3">
   689→              <Skeleton className="w-11 h-11 rounded-full" />
   690→              <div className="space-y-1.5 flex-1">
   691→                <Skeleton className="h-3.5 w-32 rounded" />
   692→                <Skeleton className="h-2.5 w-20 rounded" />
   693→              </div>
   694→            </div>
   695→            <Skeleton className="h-4 w-full rounded" />
   696→            <Skeleton className="h-4 w-3/4 rounded" />
   697→          </Card>
   698→        ))}
   699→      </div>
   700→    );
   701→  }
   702→
   703→  if (!posts || posts.length === 0) {
   704→    return (
   705→      <Card className="p-4 border-border/60 shadow-card">
   706→        <EmptyState
   707→          kind="posts"
   708→          title="پستی منتشر نشده"
   709→          description={
   710→            isSelf
   711→              ? "پست‌های شما اینجا نمایش داده می‌شوند."
   712→              : "این کاربر هنوز پستی منتشر نکرده است."
   713→          }
   714→          action={
   715→            isSelf ? (
   716→              <Button
   717→                size="sm"
   718→                variant="outline"
   719→                onClick={() => navigate({ view: "feed" })}
   720→                className="rounded-xl gap-1.5"
   721→              >
   722→                <FileText className="w-4 h-4" /> رفتن به فید
   723→              </Button>
   724→            ) : undefined
   725→          }
   726→        />
   727→      </Card>
   728→    );
   729→  }
   730→
   731→  return (
   732→    <div className="space-y-4">
   733→      {posts.map((p, i) => (
   734→        <PostCard key={p.id} post={p} index={i} />
   735→      ))}
   736→    </div>
   737→  );
   738→}
